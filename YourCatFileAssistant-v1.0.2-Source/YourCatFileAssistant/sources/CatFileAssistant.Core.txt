namespace CatFileAssistant.Core;

public enum ArchiveOperation { Create, Extract }
public enum ArchiveFormat { Zip, SevenZip, Tar, TarGZip }
public enum ArchiveLevel { Store = 0, Fast = 1, Standard = 5, Ultra = 9 }
public enum ArchiveGrouping { Combined, Separate }
public enum ArchiveOutputKind { File, Directory }

public sealed record ArchiveLimits(int MaximumEntries = 100_000, int MaximumDepth = 64,
    long MaximumBytes = 100L * 1024 * 1024 * 1024);
public sealed record ArchiveCreateOptions(ArchiveFormat Format = ArchiveFormat.Zip,
    ArchiveLevel Level = ArchiveLevel.Standard, string? Password = null, long? SplitBytes = null);
public sealed record ArchiveExtractOptions(string? Password = null, ArchiveLimits? Limits = null);

public abstract record ArchiveRequest(Guid Id, string OutputDirectory);
public sealed record CreateArchiveRequest(Guid Id, IReadOnlyList<string> SourcePaths,
    string OutputDirectory, string ArchiveName, ArchiveCreateOptions Options)
    : ArchiveRequest(Id, OutputDirectory);
public sealed record ExtractArchiveRequest(Guid Id, string SourcePath, string OutputDirectory,
    ArchiveExtractOptions Options) : ArchiveRequest(Id, OutputDirectory);
public sealed record ArchiveProgress(double? Percent, long ProcessedBytes, long? TotalBytes,
    int ProcessedEntries, int? TotalEntries, string? CurrentEntry = null, string? Stage = null);
public sealed record ArchiveStatistics(long SourceBytes, long OutputBytes, int EntryCount)
{
    public double ReductionPercent => SourceBytes > 0 ? (1d - (double)OutputBytes / SourceBytes) * 100 : 0;
}
public sealed record ArchiveInspection(string SourcePath, string? Format, long SourceBytes,
    int EntryCount, bool RequiresPassword, ConversionError? Error = null)
{
    public bool Accepted => Error is null;
}
public sealed record ArchiveResult(JobStatus Status, string? OutputPath, ArchiveOutputKind OutputKind,
    ConversionError? Error = null, ArchiveStatistics? Statistics = null);

public interface IArchiveEngine
{
    Task<ArchiveInspection> InspectAsync(string sourcePath, string? password = null,
        CancellationToken cancellationToken = default);
    Task<ArchiveResult> ExecuteAsync(ArchiveRequest request,
        IProgress<ArchiveProgress>? progress = null, CancellationToken cancellationToken = default);
}

public static class ArchiveCapabilities
{
    public static string Extension(ArchiveFormat format) => format switch
    {
        ArchiveFormat.SevenZip => ".7z", ArchiveFormat.Tar => ".tar",
        ArchiveFormat.TarGZip => ".tar.gz", _ => ".zip"
    };
    public static string Name(ArchiveFormat format) => format switch
    {
        ArchiveFormat.SevenZip => "7Z", ArchiveFormat.Tar => "TAR",
        ArchiveFormat.TarGZip => "TAR.GZ", _ => "ZIP"
    };
    public static bool SupportsPassword(ArchiveFormat format) => format is ArchiveFormat.Zip or ArchiveFormat.SevenZip;
    public static bool SupportsSplit(ArchiveFormat format) => format == ArchiveFormat.SevenZip;
    public static IReadOnlyList<ArchiveLevel> Levels(ArchiveFormat format) => format switch
    {
        ArchiveFormat.Tar => new[] { ArchiveLevel.Store },
        ArchiveFormat.TarGZip => new[] { ArchiveLevel.Fast, ArchiveLevel.Standard, ArchiveLevel.Ultra },
        _ => new[] { ArchiveLevel.Store, ArchiveLevel.Fast, ArchiveLevel.Standard, ArchiveLevel.Ultra }
    };
}

namespace CatFileAssistant.Core;

/// <summary>Brand state is derived from task outcomes, never from translated UI text.</summary>
public static class AssistantStateRules
{
    public static AssistantState ForBatch(IEnumerable<JobStatus> statuses)
    {
        var values = statuses.ToArray();
        if (values.Contains(JobStatus.Processing)) return AssistantState.PROCESSING;
        if (values.Contains(JobStatus.Failed)) return AssistantState.ERROR;
        if (values.Length > 0 && values.All(value => value == JobStatus.Completed)) return AssistantState.SUCCESS;
        return AssistantState.IDLE;
    }
}

public sealed record CompressionOptions(CompressionMode Mode, CompressionLevel Level = CompressionLevel.Medium,
    long? MaximumBytes = null, bool PreserveSourceFormat = true)
{
    public double Ratio => Level switch { CompressionLevel.Large => .8, CompressionLevel.Medium => .5, CompressionLevel.Small => .3, CompressionLevel.Tiny => .15, _ => .5 };
    public long TargetBytes(long sourceBytes) => Mode == CompressionMode.TargetSize
        ? MaximumBytes ?? 0 : (long)Math.Floor(sourceBytes * Ratio);
}

using System.Collections.Concurrent;
using System.Collections.ObjectModel;
using System.Globalization;
using System.Text;
using System.Text.Json;
using System.Text.RegularExpressions;

namespace CatFileAssistant.Core;

/// <summary>Local, deterministic application text. File names and captured template values are never translated.</summary>
public static class Locale
{
    private static readonly Regex placeholder = new(@"(?<!\{)\{(?<index>\d+)(?:,-?\d+)?(?::[^{}]*)?\}(?!\})", RegexOptions.CultureInvariant, TimeSpan.FromMilliseconds(100));
    private static string language = "zh-CN";
    private static readonly IReadOnlyDictionary<string, string> catalog = LoadCatalog();
    private static readonly IReadOnlyDictionary<string, string> reverse = new ReadOnlyDictionary<string, string>(
        catalog.GroupBy(pair => pair.Value, StringComparer.Ordinal).ToDictionary(group => group.Key, group => group.First().Key, StringComparer.Ordinal));
    private static readonly Template[] templates = catalog.Where(pair => Placeholders(pair.Key).Count != 0)
        .Select(pair => new Template(pair.Key, pair.Value)).OrderByDescending(template => template.LiteralLength).ToArray();
    private static readonly ConcurrentDictionary<string, byte> missing = new(StringComparer.Ordinal);
    private static readonly ConcurrentDictionary<(string Language, string Text), string> rendered = new();

    public static event EventHandler? Changed;
    public static string Normalize(string? value) => value?.Trim().ToLowerInvariant() is "en" or "en-us" or "english" ? "en-US" : "zh-CN";
    public static string Language
    {
        get => language;
        set
        {
            var next = Normalize(value);
            if (language == next) return;
            language = next;
            Changed?.Invoke(null, EventArgs.Empty);
        }
    }
    public static IReadOnlyDictionary<string, string> Catalog => catalog;
    public static IReadOnlyCollection<string> MissingKeys => missing.Keys.Order(StringComparer.Ordinal).ToArray();
    public static void ClearMissingKeys() { missing.Clear(); rendered.Clear(); }
    public static bool IsEnglish => Language == "en-US";

    public static string T(string source)
    {
        ArgumentNullException.ThrowIfNull(source);
        return Translate(source, IsEnglish);
    }

    private static string Translate(string source, bool english)
    {
        if (!english) return source;
        if (catalog.TryGetValue(source, out var translated)) return translated;
        if (source.Any(character => character is >= '\u4e00' and <= '\u9fff')) missing.TryAdd(source, 0);
        return source;
    }

    public static string F(FormattableString source)
    {
        ArgumentNullException.ThrowIfNull(source);
        var english = IsEnglish;
        var result = string.Format(english ? CultureInfo.GetCultureInfo("en-US") : CultureInfo.GetCultureInfo("zh-CN"), Translate(source.Format, english), source.GetArguments());
        if (english)
        {
            if (rendered.Count > 4096) rendered.Clear();
            // A translated sentence can contain a user's Chinese filename. It is
            // already localized and must neither translate that capture nor report it.
            rendered.TryAdd(("en-US", result), result);
        }
        return result;
    }

    /// <summary>Translates an already rendered application label in either direction. Captures remain byte-for-byte intact.</summary>
    public static string TranslateText(string text)
    {
        if (string.IsNullOrEmpty(text)) return text;
        if (rendered.Count > 4096) rendered.Clear();
        return rendered.GetOrAdd((Language, text), key => TranslateRendered(key.Text, key.Language == "en-US"));
    }

    private static string TranslateRendered(string text, bool english)
    {
        if ((english ? catalog : reverse).TryGetValue(text, out var translated)) return translated;
        if (english && reverse.ContainsKey(text)) return text;
        // Most presentation updates already passed through T/F. Avoid scanning every
        // template for plain English or source-language labels on animation/progress ticks.
        var containsChinese = text.Any(character => character is >= '\u4e00' and <= '\u9fff');
        if (english && !containsChinese) return text;
        foreach (var template in templates)
        {
            if (template.TryTranslate(text, english, out translated)) return translated;
        }
        if (english && containsChinese) missing.TryAdd(text, 0);
        return text;
    }

    /// <summary>Checks catalog placeholders without invoking formatters or changing the active language.</summary>
    public static IReadOnlyList<string> ValidateCatalog()
    {
        var errors = new List<string>();
        foreach (var pair in catalog)
        {
            var source = Placeholders(pair.Key).Order().ToArray();
            var target = Placeholders(pair.Value).Order().ToArray();
            if (!source.SequenceEqual(target)) errors.Add("Placeholder mismatch: " + pair.Key);
            if (string.IsNullOrWhiteSpace(pair.Value)) errors.Add("Empty translation: " + pair.Key);
            try { _ = CompositeFormat.Parse(pair.Key); _ = CompositeFormat.Parse(pair.Value); }
            catch (FormatException) { errors.Add("Invalid format: " + pair.Key); }
        }
        return errors;
    }

    private static IReadOnlyDictionary<string, string> LoadCatalog()
    {
        using var stream = typeof(Locale).Assembly.GetManifestResourceStream("CatFileAssistant.Core.Localization.en-US.json")
            ?? throw new InvalidOperationException("Application language catalog is missing.");
        return new ReadOnlyDictionary<string, string>(JsonSerializer.Deserialize<Dictionary<string, string>>(stream)
            ?? throw new InvalidOperationException("Application language catalog is empty."));
    }

    private static List<int> Placeholders(string value) => Regex.Matches(value, @"(?<!\{)\{(?<index>\d+)(?:,-?\d+)?(?::[^{}]*)?\}(?!\})", RegexOptions.CultureInvariant)
        .Select(match => int.Parse(match.Groups["index"].Value, CultureInfo.InvariantCulture)).ToList();

    private sealed class Template
    {
        private readonly string source, target;
        private readonly Regex sourcePattern, targetPattern;
        public int LiteralLength { get; }
        public Template(string source, string target)
        {
            this.source = source; this.target = target;
            sourcePattern = Pattern(source); targetPattern = Pattern(target);
            LiteralLength = placeholder.Replace(source, "").Length;
        }
        public bool TryTranslate(string text, bool english, out string translated)
        {
            translated = text;
            Match match;
            try { match = (english ? sourcePattern : targetPattern).Match(text); }
            catch (RegexMatchTimeoutException) { return false; }
            if (!match.Success) return false;
            translated = placeholder.Replace(english ? target : source, part => match.Groups["p" + part.Groups["index"].Value].Value)
                .Replace("{{", "{", StringComparison.Ordinal).Replace("}}", "}", StringComparison.Ordinal);
            return true;
        }
        private static Regex Pattern(string format)
        {
            var result = new StringBuilder("\\A");
            var position = 0;
            var seen = new HashSet<string>();
            foreach (Match match in placeholder.Matches(format))
            {
                result.Append(Regex.Escape(format[position..match.Index].Replace("{{", "{", StringComparison.Ordinal).Replace("}}", "}", StringComparison.Ordinal)));
                var index = match.Groups["index"].Value;
                result.Append(seen.Add(index) ? "(?<p" + index + ">.*?)" : "\\k<p" + index + ">");
                position = match.Index + match.Length;
            }
            result.Append(Regex.Escape(format[position..].Replace("{{", "{", StringComparison.Ordinal).Replace("}}", "}", StringComparison.Ordinal))).Append("\\z");
            return new Regex(result.ToString(), RegexOptions.CultureInvariant | RegexOptions.Singleline, TimeSpan.FromMilliseconds(30));
        }
    }
}

/// <summary>Original numeric pattern rules v1; no external or leaked password collection is used.</summary>
public static class RecoveryCandidates
{
    public static int SpaceSize(int digits) => digits switch
    { 4 => 10_000, 6 => 1_000_000, _ => throw new ArgumentException(Locale.T("请选择 4 位或 6 位纯数字密码。")) };

    public static int CpuPercent(RecoveryIntensity intensity) => intensity switch
    { RecoveryIntensity.Low => 20, RecoveryIntensity.Medium => 50, RecoveryIntensity.High => 80,
        _ => throw new ArgumentException(Locale.T("请选择有效的计算强度。")) };

    public static IReadOnlyList<string> NormalizeNumbers(IEnumerable<string>? inputs)
    {
        var values = new List<string>(); var seen = new HashSet<string>(StringComparer.Ordinal);
        foreach (var input in inputs ?? [])
        {
            var value = input?.Trim() ?? "";
            if (value.Length is < 1 or > 20 || value.Any(c => c is < '0' or > '9'))
                throw new ArgumentException(Locale.T("每条常用数字须为 1–20 个数字（0–9）。"));
            if (seen.Add(value)) values.Add(value);
            if (values.Count > 100) throw new ArgumentException(Locale.T("常用数字最多保存 100 条。"));
        }
        return values.AsReadOnly();
    }

    public static IReadOnlyList<IReadOnlyList<string>> CommonStages(IEnumerable<string>? inputs, int digits)
    {
        _ = SpaceSize(digits);
        var numbers = NormalizeNumbers(inputs);
        var seen = new HashSet<string>(StringComparer.Ordinal);
        var exact = new List<string>(); var variants = new List<string>(); var builtIn = new List<string>();
        void Add(List<string> target, string candidate) { if (seen.Add(candidate)) target.Add(candidate); }
        foreach (var value in numbers.Where(value => value.Length == digits)) Add(exact, value);
        foreach (var value in numbers)
        {
            if (value.Length > digits) { Add(variants, value[..digits]); Add(variants, value[^digits..]); }
            else if (value.Length < digits)
            {
                Add(variants, value.PadLeft(digits, '0')); Add(variants, value.PadRight(digits, '0'));
                Add(variants, string.Concat(Enumerable.Repeat(value, (digits + value.Length - 1) / value.Length))[..digits]);
            }
        }
        foreach (var candidate in BuiltIn(digits)) Add(builtIn, candidate);
        return new IReadOnlyList<string>[] { exact.AsReadOnly(), variants.AsReadOnly(), builtIn.AsReadOnly() };
    }

    public static IReadOnlyList<string> BuiltIn(int digits)
    {
        _ = SpaceSize(digits); var values = new List<string>();
        for (var digit = 0; digit < 10; digit++) values.Add(new string((char)('0' + digit), digits));
        foreach (var direction in new[] { 1, -1 })
            for (var start = 0; start < 10; start++)
                values.Add(new string(Enumerable.Range(0, digits).Select(i => (char)('0' + (start + direction * i + 20) % 10)).ToArray()));
        foreach (var pair in new[] { "12", "21", "10", "01", "52", "25", "69", "96" })
            values.Add(string.Concat(Enumerable.Repeat(pair, digits / 2)));
        return values.Distinct(StringComparer.Ordinal).ToArray();
    }

    public static long Total(RecoveryOptions options, IEnumerable<string>? numbers = null)
    {
        if (!Enum.IsDefined(options.Mode)) throw new ArgumentException(Locale.T("请选择有效的恢复方式。"));
        _ = CpuPercent(options.Intensity);
        return options.Mode == RecoveryMode.Cpu ? SpaceSize(options.Digits) : CommonStages(numbers, options.Digits).Sum(s => (long)s.Count);
    }

    public static IEnumerable<string> EnumerateDigits(int digits)
    {
        var count = SpaceSize(digits); var format = "D" + digits;
        for (var i = 0; i < count; i++) yield return i.ToString(format, CultureInfo.InvariantCulture);
    }

    public int FailedCount => Results.Count(r => r.Result.Status == JobStatus.Failed);

public sealed class SerialArchiveQueue(IArchiveEngine engine)
{
    public async Task<ArchiveBatchResult> RunAsync(IReadOnlyList<ArchiveRequest> requests,
        Action<ArchiveRequest>? onStarted = null, Action<ArchiveRequest, ArchiveResult>? onCompleted = null,
        Func<ArchiveRequest, IProgress<ArchiveProgress>?>? progressFactory = null,
        Action<Exception>? onObserverError = null, CancellationToken cancellationToken = default)
    {
        var results = new List<ArchiveTaskResult>();
        var cancelled = false;
        foreach (var request in requests)
        {
            if (cancellationToken.IsCancellationRequested) { cancelled = true; break; }
            Notify(() => onStarted?.Invoke(request), onObserverError);
            IProgress<ArchiveProgress>? progress = null;
            Notify(() => progress = progressFactory?.Invoke(request), onObserverError);
            ArchiveResult result;
            try { result = await engine.ExecuteAsync(request, progress, cancellationToken); }
            catch (OperationCanceledException) { result = new(JobStatus.Cancelled, null, ArchiveOutputKind.File); }
            catch (Exception)
            {
                result = new(JobStatus.Failed, null, ArchiveOutputKind.File,
                    new("archive.unexpected", Locale.T("压缩包处理遇到了问题。"), Locale.T("请确认文件可读取，并重试。")));
            }
            results.Add(new(request, result));
            Notify(() => onCompleted?.Invoke(request, result), onObserverError);
            if (result.Status == JobStatus.Cancelled || cancellationToken.IsCancellationRequested) { cancelled = true; break; }
        }
        return new(results.AsReadOnly(), cancelled);
    }
    private static void Notify(Action callback, Action<Exception>? observer)
    {
        try { callback(); } catch (Exception error) { try { observer?.Invoke(error); } catch (Exception) { } }
    }
}

/// <summary>Portable sequential task orchestration. No UI types, file deletion, or translated status text.</summary>
public sealed class SerialConversionQueue(IConversionEngine engine)
{
    public async Task<QueueBatchResult> RunAsync(
        IReadOnlyList<ConversionRequest> requests,
        Action<ConversionRequest>? onStarted = null,
        Action<ConversionRequest, ConversionResult>? onCompleted = null,
        Func<ConversionRequest, IProgress<ConversionProgress>?>? progressFactory = null,
        Action<Exception>? onObserverError = null,
        CancellationToken cancellationToken = default)
    {
        var outcomes = new List<QueueTaskResult>();
        var cancelled = false;
        foreach (var request in requests)
        {
            if (cancellationToken.IsCancellationRequested) { cancelled = true; break; }
            Notify(() => onStarted?.Invoke(request), onObserverError);
            IProgress<ConversionProgress>? progress = null;
            Notify(() => progress = progressFactory?.Invoke(request), onObserverError);
            ConversionResult result;
            try { result = await engine.ConvertAsync(request, progress, cancellationToken); }
            catch (OperationCanceledException) { result = new(JobStatus.Cancelled, null, null); }
            catch (ConversionEngineException exception) { result = new(JobStatus.Failed, null, exception.Error); }
            catch (Exception exception)
            {
                result = new(JobStatus.Failed, null, new ConversionError("ENGINE_UNEXPECTED", Locale.T("这个文件好像遇到了一点问题。"), Locale.T("请重试，或换一种输出格式。"), TechnicalDetails: exception.ToString()));
            }
            outcomes.Add(new QueueTaskResult(request, result));
            Notify(() => onCompleted?.Invoke(request, result), onObserverError);
            if (result.Status == JobStatus.Cancelled || cancellationToken.IsCancellationRequested) { cancelled = true; break; }
        }
        return new QueueBatchResult(outcomes.AsReadOnly(), cancelled);
    }

    private static void Notify(Action notification, Action<Exception>? onObserverError)
    {
        try { notification(); }
        catch (Exception exception)
        {
            // An observer cannot make an already committed output a failed conversion.
            try { onObserverError?.Invoke(exception); }
            catch (Exception) { }
        }
    }
}
