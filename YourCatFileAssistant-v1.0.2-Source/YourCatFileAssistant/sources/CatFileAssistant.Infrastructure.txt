using System.Buffers.Binary;
using CatFileAssistant.Core;

namespace CatFileAssistant.Infrastructure;

/// <summary>Bounded header parsing and fixed-length timing repairs; FFmpeg owns all pixel decoding.</summary>
internal static class AnimationMetadata
{
    internal sealed record Inspection(AnimationInfo? Animation, bool HasAlpha, int Width = 0, int Height = 0, double SampleAspectRatio = 1);

    // Conversion validation must observe encoded delays, including zero, without player defaults.
    public static Inspection ReadGifRaw(string path)
    {
        using var file = File.OpenRead(path);
        return ReadGif(file, rawTiming: true);
    }

    public static Inspection Read(string path)
    {
        using var file = File.OpenRead(path);
        Span<byte> signature = stackalloc byte[12];
        if (file.Read(signature) < 12) return new(null, false);
        file.Position = 0;
        if (signature[..4].SequenceEqual("RIFF"u8) && signature[8..12].SequenceEqual("WEBP"u8)) return ReadWebP(file);
        if (signature[..6].SequenceEqual("GIF87a"u8) || signature[..6].SequenceEqual("GIF89a"u8)) return ReadGif(file);
        if (signature[..8].SequenceEqual(new byte[] { 137, 80, 78, 71, 13, 10, 26, 10 })) return ReadPng(file);
        return new(null, false);
    }

    private static Inspection ReadWebP(Stream file)
    {
        var delays = new List<int>();
        var loop = 1;
        var alpha = false;
        Span<byte> bytes = stackalloc byte[3];
        foreach (var chunk in WebPChunks(file))
        {
            file.Position = chunk.Payload;
            if (chunk.Tag == "VP8X" && chunk.Length >= 10) alpha = (file.ReadByte() & 16) != 0;
            if (chunk.Tag == "VP8L" && chunk.Length >= 5)
            {
                file.Position += 4;
                alpha |= (file.ReadByte() & 16) != 0;
            }
            if (chunk.Tag == "ANIM" && chunk.Length >= 6)
            {
                file.Position += 4;
                file.ReadExactly(bytes[..2]);
                loop = BinaryPrimitives.ReadUInt16LittleEndian(bytes[..2]);
            }
            if (chunk.Tag == "ANMF" && chunk.Length >= 16)
            {
                file.Position += 12;
                file.ReadExactly(bytes);
                var delay = UInt24(bytes);
                delays.Add(delay == 0 ? 100 : delay);
            }
        }
        return new(delays.Count > 0 ? new(delays.Count, delays.Sum(value => (long)value), loop, delays.AsReadOnly()) : null, alpha);
    }

    private static Inspection ReadPng(Stream file)
    {
        file.Position = 8;
        Span<byte> chunk = stackalloc byte[8];
        Span<byte> header = stackalloc byte[10];
        var alpha = false;
        var count = 0;
        while (file.Position + 8 <= file.Length && ++count <= 10000)
        {
            file.ReadExactly(chunk);
            var length = BinaryPrimitives.ReadUInt32BigEndian(chunk[..4]);
            var next = checked(file.Position + length + 4L);
            if (next > file.Length) throw new InvalidDataException(Locale.T("PNG 块长度无效。"));
            if (chunk[4..].SequenceEqual("IHDR"u8) && length >= 13)
            {
                file.ReadExactly(header);
                alpha |= header[9] is 4 or 6;
            }
            if (chunk[4..].SequenceEqual("tRNS"u8)) alpha = true;
            if (chunk[4..].SequenceEqual("IDAT"u8) || chunk[4..].SequenceEqual("IEND"u8)) break;
            file.Position = next;
        }
        return new(null, alpha);
    }

    private static Inspection ReadGif(Stream file, bool rawTiming = false)
    {
        Span<byte> header = stackalloc byte[13]; file.ReadExactly(header);
        if (!header[..6].SequenceEqual("GIF87a"u8) && !header[..6].SequenceEqual("GIF89a"u8))
            throw new InvalidDataException(Locale.T("GIF 文件头无效。"));
        var width = BinaryPrimitives.ReadUInt16LittleEndian(header[6..8]);
        var height = BinaryPrimitives.ReadUInt16LittleEndian(header[8..10]);
        if (width == 0 || height == 0) throw new InvalidDataException(Locale.T("GIF 画布尺寸无效。"));
        if ((header[10] & 128) != 0) Skip(file, 3L * (1 << ((header[10] & 7) + 1)));
        var delays = new List<int>();
        var pendingDelay = 100;
        var alpha = false;
        var loop = -1;
        var trailer = false;
        Span<byte> control = stackalloc byte[5];
        Span<byte> loopData = stackalloc byte[3];
        Span<byte> image = stackalloc byte[9];
        while (file.Position < file.Length)
        {
            var marker = file.ReadByte();
            if (marker == 0x3b) { trailer = true; break; }
            if (marker == 0x21)
            {
                var label = file.ReadByte();
                if (label == 0xf9)
                {
                    if (file.ReadByte() != 4) throw new InvalidDataException(Locale.T("GIF 图形控制块无效。"));
                    file.ReadExactly(control);
                    alpha |= (control[0] & 1) != 0;
                    pendingDelay = BinaryPrimitives.ReadUInt16LittleEndian(control[1..3]) * 10;
                    if (pendingDelay == 0 && !rawTiming) pendingDelay = 100;
                    if (control[4] != 0) throw new InvalidDataException(Locale.T("GIF 控制块终止标记无效。"));
                }
                else if (label == 0xff)
                {
                    var count = file.ReadByte();
                    if (count < 0) throw new EndOfStreamException();
                    var name = new byte[count]; file.ReadExactly(name);
                    var netscape = name.AsSpan().SequenceEqual("NETSCAPE2.0"u8) || name.AsSpan().SequenceEqual("ANIMEXTS1.0"u8);
                    var block = file.ReadByte();
                    if (block < 0) throw new EndOfStreamException();
                    if (netscape && block == 3)
                    {
                        file.ReadExactly(loopData);
                        if (loopData[0] == 1) loop = BinaryPrimitives.ReadUInt16LittleEndian(loopData[1..]);
                    }
                    else Skip(file, block);
                    if (block != 0) SkipSubBlocks(file);
                }
                else SkipSubBlocks(file);
            }
            else if (marker == 0x2c)
            {
                file.ReadExactly(image);
                if (rawTiming)
                {
                    var left = BinaryPrimitives.ReadUInt16LittleEndian(image[..2]);
                    var top = BinaryPrimitives.ReadUInt16LittleEndian(image[2..4]);
                    var frameWidth = BinaryPrimitives.ReadUInt16LittleEndian(image[4..6]);
                    var frameHeight = BinaryPrimitives.ReadUInt16LittleEndian(image[6..8]);
                    if (frameWidth == 0 || frameHeight == 0 || left + frameWidth > width || top + frameHeight > height)
                        throw new InvalidDataException(Locale.T("GIF 动画帧超出了画布。"));
                }
                if ((image[8] & 128) != 0) Skip(file, 3L * (1 << ((image[8] & 7) + 1)));
                if (file.ReadByte() < 0) throw new EndOfStreamException();
                SkipSubBlocks(file);
                delays.Add(pendingDelay); pendingDelay = 100;
            }
            else throw new InvalidDataException(Locale.T("GIF 块无效。"));
            if (delays.Count > 1_000_000) throw new InvalidDataException(Locale.T("动画帧数过多。"));
        }
        if (rawTiming && (!trailer || file.Position != file.Length)) throw new InvalidDataException(Locale.T("GIF 文件未完整结束。"));
        return new(delays.Count > 0 ? new(delays.Count, delays.Sum(value => (long)value), loop, delays.AsReadOnly(), true) : null, alpha, width, height,
            header[12] == 0 ? 1d : (header[12] + 15d) / 64);
    }

    public static void RepairWebP(string path, AnimationInfo original, bool independentFrames = false)
    {
        using var file = new FileStream(path, FileMode.Open, FileAccess.ReadWrite, FileShare.None);
        var chunks = WebPChunks(file).ToArray();
        var frames = chunks.Where(value => value.Tag == "ANMF").ToArray();
        if (frames.Length == 0) throw new InvalidDataException(Locale.T("动画输出意外变为静态图片。"));
        long preceding = 0;
        Span<byte> data = stackalloc byte[3];
        for (var i = 0; i < frames.Length; i++)
        {
            file.Position = frames[i].Payload + 12; file.ReadExactly(data);
            if (i < frames.Length - 1) preceding += UInt24(data);
            if (independentFrames)
            {
                // The normal WebP encoder emits complete canvas frames. Replace, rather than blend,
                // so partially transparent frames cannot accumulate alpha over previous frames.
                file.Position = frames[i].Payload + 15; file.WriteByte(2);
            }
        }
        var last = original.DurationMilliseconds - preceding;
        if (last <= 0 || last > 0xffffff) throw new InvalidDataException(Locale.T("动画时长无法安全保留。"));
        file.Position = frames[^1].Payload + 12; WriteUInt24(data, (int)last); file.Write(data);
        var loop = original.IsGif ? original.LoopCount switch { -1 => 1, 0 => 0, _ => Math.Min(65535, original.LoopCount + 1) } : original.LoopCount;
        var animation = chunks.FirstOrDefault(value => value.Tag == "ANIM");
        if (animation.Tag is null) throw new InvalidDataException(Locale.T("动画循环信息丢失。"));
        file.Position = animation.Payload + 4;
        Span<byte> loopBytes = stackalloc byte[2]; BinaryPrimitives.WriteUInt16LittleEndian(loopBytes, (ushort)loop); file.Write(loopBytes);
    }

    public static int GifLoop(AnimationInfo source) => source.IsGif ? source.LoopCount
        : source.LoopCount switch { 0 => 0, 1 => -1, _ => source.LoopCount - 1 };

    private readonly record struct Chunk(string? Tag, long Payload, uint Length);
    private static IEnumerable<Chunk> WebPChunks(Stream file)
    {
        file.Position = 12;
        var bytes = new byte[8];
        var count = 0;
        while (file.Position + 8 <= file.Length)
        {
            file.ReadExactly(bytes);
            var length = BinaryPrimitives.ReadUInt32LittleEndian(bytes.AsSpan(4));
            var payload = file.Position;
            var next = checked(payload + length + (length & 1));
            if (next > file.Length || ++count > 1_000_000) throw new InvalidDataException(Locale.T("WebP 块长度无效。"));
            yield return new(System.Text.Encoding.ASCII.GetString(bytes, 0, 4), payload, length);
            file.Position = next;
        }
    }
    private static void SkipSubBlocks(Stream stream)
    {
        int count;
        while ((count = stream.ReadByte()) > 0) Skip(stream, count);
        if (count < 0) throw new EndOfStreamException();
    }
    private static void Skip(Stream stream, long count)
    {
        if (count < 0 || stream.Position + count > stream.Length) throw new EndOfStreamException();
        stream.Position += count;
    }
    private static int UInt24(ReadOnlySpan<byte> bytes) => bytes[0] | bytes[1] << 8 | bytes[2] << 16;
    private static void WriteUInt24(Span<byte> bytes, int value) { bytes[0] = (byte)value; bytes[1] = (byte)(value >> 8); bytes[2] = (byte)(value >> 16); }
}

using System.ComponentModel;
using CatFileAssistant.Core;
using CatFileAssistant.Ownership;

namespace CatFileAssistant.Infrastructure;

/// <summary>Local archive work executes in an isolated worker; that worker alone owns staging and commit.</summary>
public sealed class ArchiveEngine : IArchiveEngine
{
    private readonly DiagnosticLog? log;
    public string DllPath { get; }
    public string WorkerPath { get; }
    public bool IsReady => File.Exists(DllPath) && File.Exists(WorkerPath);

    public ArchiveEngine(string? dllPath = null, string? workerPath = null, DiagnosticLog? log = null)
    {
        DllPath = string.IsNullOrWhiteSpace(dllPath) ? FindDefaultDll() : Path.GetFullPath(dllPath);
        WorkerPath = string.IsNullOrWhiteSpace(workerPath) ? FindDefaultWorker() : Path.GetFullPath(workerPath);
        this.log = log;
    }

    public async Task<ArchiveInspection> InspectAsync(string sourcePath, string? password = null,
        CancellationToken cancellationToken = default)
    {
        cancellationToken.ThrowIfCancellationRequested();
        try
        {
            sourcePath = Path.GetFullPath(sourcePath);
            if (!File.Exists(sourcePath)) return new(sourcePath, null, 0, 0, false,
                new("archive.input-missing", Locale.T("没有找到压缩包或首卷。"), Locale.T("请确认压缩包和所有分卷都保留在同一个文件夹中。")));
            if (!IsReady) return new(sourcePath, null, 0, 0, false, MissingEngine());
            var outcome = await ArchiveWorkerProcess.RunAsync(WorkerPath, DllPath,
                new ArchiveWorkerRequest { Action = "inspect", Id = Guid.NewGuid(), SourcePath = sourcePath,
                    Password = EmptyToNull(password), Limits = WorkerLimits(new ArchiveLimits()) },
                null, cancellationToken).ConfigureAwait(false);
            var result = outcome.Result;
            if (result?.Status == "completed")
                return new(sourcePath, result.Format, result.SourceBytes, result.EntryCount, result.RequiresPassword);
            if (outcome.CancellationRequested || result?.Status == "cancelled") throw new OperationCanceledException(cancellationToken);
            return new(sourcePath, result?.Format, Math.Max(0, result?.SourceBytes ?? 0),
                Math.Max(0, result?.EntryCount ?? 0), result?.RequiresPassword ?? false, ErrorFor(outcome, password));
        }
        catch (OperationCanceledException) { throw; }
        catch (Exception exception)
        {
            return new(sourcePath, null, 0, 0, false, ExceptionError(exception));
        }
    }

    public async Task<ArchiveResult> ExecuteAsync(ArchiveRequest request,
        IProgress<ArchiveProgress>? progress = null, CancellationToken cancellationToken = default)
    {
        string? workDirectory = null;
        string? journalPath = null;
        string? ownedOutputDirectory = null;
        var kind = request is ExtractArchiveRequest || request is CreateArchiveRequest { Options.SplitBytes: not null }
            ? ArchiveOutputKind.Directory : ArchiveOutputKind.File;
        try
        {
            cancellationToken.ThrowIfCancellationRequested();
            if (!IsReady) return new(JobStatus.Failed, null, kind, MissingEngine());
            var invalid = ValidateRequest(request);
            if (invalid is not null) return new(JobStatus.Failed, null, kind, invalid);
            var outputDirectory = Path.GetFullPath(request.OutputDirectory);
            ownedOutputDirectory = outputDirectory;
            do { workDirectory = Path.Combine(outputDirectory, $".cat-archive-{request.Id:N}-{Guid.NewGuid():N}"); }
            while (Directory.Exists(workDirectory) || File.Exists(workDirectory));
            journalPath = OwnedArtifacts.CreateArchiveJournal(workDirectory);
            // Only the worker creates this directory, after freezing a source-directory input plan.
            var workerRequest = ToWorkerRequest(request, workDirectory, journalPath);
            log?.Write($"Archive operation started: task={request.Id:N}, action={workerRequest.Action}");
            var outcome = await ArchiveWorkerProcess.RunAsync(WorkerPath, DllPath, workerRequest, update =>
            {
                try
                {
                    progress?.Report(new ArchiveProgress(update.Percent is { } percent ? Math.Clamp(percent, 0, 100) : null,
                        update.ProcessedBytes, update.TotalBytes, update.ProcessedEntries, update.TotalEntries,
                        update.CurrentEntry, update.Stage));
                }
                catch (Exception) { }
            }, cancellationToken).ConfigureAwait(false);
            var result = outcome.Result;
            if (result?.Status == "completed")
            {
                var actualKind = result.OutputKind == "directory" ? ArchiveOutputKind.Directory : ArchiveOutputKind.File;
                if (string.IsNullOrWhiteSpace(result.OutputPath)
                    || !Path.IsPathFullyQualified(result.OutputPath) || actualKind != kind
                    || !string.Equals(Path.GetDirectoryName(Path.GetFullPath(result.OutputPath)),
                        Path.TrimEndingDirectorySeparator(outputDirectory), StringComparison.OrdinalIgnoreCase)
                    || (actualKind == ArchiveOutputKind.Directory ? !Directory.Exists(result.OutputPath) : !File.Exists(result.OutputPath)))
                    return new(JobStatus.Failed, null, kind,
                        new("archive.output-missing", Locale.T("压缩包引擎没有返回完整结果。"), Locale.T("请确认保存位置可用，重试此任务。")));
                log?.Write($"Archive operation completed: task={request.Id:N}");
                return new(JobStatus.Completed, result.OutputPath, actualKind, Statistics:
                    new(result.SourceBytes, result.OutputBytes, result.EntryCount));
            }
            if (outcome.CancellationRequested || result?.Status == "cancelled") return new(JobStatus.Cancelled, null, kind);
            return new(JobStatus.Failed, null, kind, ErrorFor(outcome, workerRequest.Password));
        }
        catch (OperationCanceledException) { return new(JobStatus.Cancelled, null, kind); }
        catch (Exception exception) { return new(JobStatus.Failed, null, kind, ExceptionError(exception)); }
        finally
        {
            if (workDirectory is not null && ownedOutputDirectory is not null)
            {
                try
                {
                    if (!string.Equals(Path.GetDirectoryName(Path.GetFullPath(workDirectory)),
                        Path.TrimEndingDirectorySeparator(Path.GetFullPath(ownedOutputDirectory)), StringComparison.OrdinalIgnoreCase))
                        throw new IOException("Temporary directory is not a direct child of the selected output folder.");
                    if (journalPath is not null && !ArchiveTemporaryJournal.Cleanup(journalPath))
                        log?.Write("Archive receipt retained files which could not be safely removed.");
                }
                catch (IOException) { log?.Write("Archive temporary cleanup could not complete."); }
                catch (UnauthorizedAccessException) { log?.Write("Archive temporary cleanup was denied."); }
            }
        }
    }

    private static ArchiveWorkerRequest ToWorkerRequest(ArchiveRequest request, string workDirectory, string journalPath) => request switch
    {
        CreateArchiveRequest create => new ArchiveWorkerRequest
        {
            Action = "create", Id = create.Id,
            SourcePaths = NormalizeCreateSources(create.SourcePaths),
            OutputDirectory = Path.GetFullPath(create.OutputDirectory), WorkDirectory = workDirectory, JournalPath = journalPath,
            ArchiveName = create.ArchiveName,
            Format = create.Options.Format switch { ArchiveFormat.SevenZip => "7z", ArchiveFormat.Tar => "tar",
                ArchiveFormat.TarGZip => "tar.gz", _ => "zip" },
            Level = (int)create.Options.Level, Password = EmptyToNull(create.Options.Password),
            SplitBytes = create.Options.SplitBytes, Limits = WorkerLimits(new ArchiveLimits())
        },
        ExtractArchiveRequest extract => new ArchiveWorkerRequest
        {
            Action = "extract", Id = extract.Id, SourcePath = Path.GetFullPath(extract.SourcePath),
            OutputDirectory = Path.GetFullPath(extract.OutputDirectory), WorkDirectory = workDirectory, JournalPath = journalPath,
            Password = EmptyToNull(extract.Options.Password),
            Limits = WorkerLimits(extract.Options.Limits ?? new ArchiveLimits())
        },
        _ => throw new ArgumentException(Locale.T("未知压缩包任务。"), nameof(request))
    };

    private static ConversionError? ValidateRequest(ArchiveRequest request)
    {
        if (string.IsNullOrWhiteSpace(request.OutputDirectory)) return Invalid(Locale.T("请选择保存文件夹。"));
        if (request is CreateArchiveRequest create)
        {
            if (create.SourcePaths.Count == 0) return Invalid(Locale.T("请先添加文件或文件夹。"));
            if (string.IsNullOrWhiteSpace(create.ArchiveName)
                || create.ArchiveName.IndexOfAny(Path.GetInvalidFileNameChars()) >= 0
                || create.ArchiveName is "." or ".." || create.ArchiveName.EndsWith('.') || create.ArchiveName.EndsWith(' '))
                return Invalid(Locale.T("请输入有效的压缩包名称。"));
            if (!Enum.IsDefined(create.Options.Format) || !Enum.IsDefined(create.Options.Level)
                || !ArchiveCapabilities.Levels(create.Options.Format).Contains(create.Options.Level))
                return Invalid(Locale.T("请选择此格式支持的压缩档位。"));
            if (!string.IsNullOrEmpty(create.Options.Password) && !ArchiveCapabilities.SupportsPassword(create.Options.Format))
                return Invalid(Locale.T("只有 ZIP 和 7Z 支持密码保护。"));
            if (create.Options.Format == ArchiveFormat.Zip && create.Options.Password is { Length: > 0 } zipPassword
                && (zipPassword.Length > 99 || zipPassword.Any(character => character is < ' ' or > '~')))
                return new("archive.zip-password", Locale.T("ZIP 密码须为 1–99 个可打印英文字符。"),
                    Locale.T("需要中文或 emoji 密码时，请选择 7Z 格式。"));
            if (create.Options.SplitBytes is { } split && (split <= 0 || !ArchiveCapabilities.SupportsSplit(create.Options.Format)))
                return Invalid(Locale.T("请选择 7Z 和有效的分卷大小。"));
        }
        else if (request is ExtractArchiveRequest extract)
        {
            if (string.IsNullOrWhiteSpace(extract.SourcePath)) return Invalid(Locale.T("请选择压缩包。"));
            var limits = extract.Options.Limits ?? new ArchiveLimits();
            if (limits.MaximumEntries <= 0 || limits.MaximumDepth <= 0 || limits.MaximumBytes <= 0)
                return Invalid(Locale.T("请填写大于零的安全上限。"));
        }
        else return Invalid(Locale.T("这个压缩包任务无效。"));
        return null;
    }

    private ConversionError ErrorFor(ArchiveWorkerOutcome outcome, string? password)
    {
        var error = outcome.Result?.Error;
        var code = string.IsNullOrWhiteSpace(error?.Code) ? "archive.worker-failed" : error.Code;
        var message = string.IsNullOrWhiteSpace(error?.Message) ? Locale.T("压缩包引擎未能完成此任务。") : error.Message;
        var suggestion = string.IsNullOrWhiteSpace(error?.Suggestion)
            ? Locale.T("请检查文件是否完整、所有分卷是否齐全，以及保存位置的剩余空间。") : error.Suggestion;
        var id = log?.Write($"Archive worker failed: exit={outcome.ExitCode}, protocol={outcome.ProtocolError ?? "none"}");
        // Native error text and the original request are deliberately excluded from the log.
        return new(ArchiveWorkerProcess.Redact(code, password), ArchiveWorkerProcess.Redact(message, password),
            ArchiveWorkerProcess.Redact(suggestion, password), id);
    }

    private ConversionError ExceptionError(Exception exception)
    {
        var id = log?.Write("Archive parent operation failed: " + exception.GetType().Name);
        return exception switch
        {
            UnauthorizedAccessException => new("archive.access", Locale.T("没有读取或保存文件的权限。"), Locale.T("请选择你有读写权限的本机文件夹。"), id),
            IOException => new("archive.io", Locale.T("读取文件或保存结果时遇到了问题。"), Locale.T("请检查文件占用、剩余空间和保存位置。"), id),
            ArgumentException => new("archive.path", Locale.T("文件路径或归档设置无效。"), Locale.T("请重新选择文件与保存位置。"), id),
            Win32Exception => new("archive.worker-start", Locale.T("压缩包引擎无法启动。"), Locale.T("请完整解压软件包并检查引擎 DLL。"), id),
            _ => new("archive.unexpected", Locale.T("压缩包处理遇到了问题。"), Locale.T("请重试；诊断记录不包含密码。"), id)
        };
    }

    private static ConversionError Invalid(string message) => new("archive.options", message, Locale.T("调整设置后再开始。"));
    private static ConversionError MissingEngine() => new("archive.engine-missing", Locale.T("没有找到完整的压缩包引擎。"),
        Locale.T("请完整解压软件包，保留 worker 文件夹及 tools/7zip/bin/7z.dll；也可在设置中选择兼容的 DLL。"));
    private static ArchiveWorkerLimits WorkerLimits(ArchiveLimits limits) => new(limits.MaximumEntries, limits.MaximumDepth, limits.MaximumBytes);
    private static string? EmptyToNull(string? value) => string.IsNullOrEmpty(value) ? null : value;

    private static IReadOnlyList<string> NormalizeCreateSources(IEnumerable<string> sources)
    {
        var unique = sources.Select(path => Path.TrimEndingDirectorySeparator(Path.GetFullPath(path)))
            .Distinct(StringComparer.OrdinalIgnoreCase).ToArray();
        return unique.Where(path => !unique.Any(other => !string.Equals(path, other, StringComparison.OrdinalIgnoreCase)
            && Directory.Exists(other) && path.StartsWith(Path.EndsInDirectorySeparator(other) ? other
                : other + Path.DirectorySeparatorChar, StringComparison.OrdinalIgnoreCase))).ToArray();
    }


    private static string FindDefaultDll()
    {
        var packaged = Path.Combine(AppContext.BaseDirectory, "tools", "7zip", "bin", "7z.dll");
        if (File.Exists(packaged)) return packaged;
        foreach (var parent in WorkspaceParents())
        {
            var candidate = Path.Combine(parent, "tools", "7zip", "bin", "7z.dll");
            if (File.Exists(candidate)) return candidate;
        }
        return packaged;
    }

    private static string FindDefaultWorker()
    {
        const string executable = "CatFileAssistant.ArchiveWorker.exe";
        var packaged = Path.Combine(AppContext.BaseDirectory, "worker", executable);
        if (File.Exists(packaged)) return packaged;
        foreach (var parent in WorkspaceParents())
        {
            foreach (var folder in new[] { Path.Combine(parent, "src", "CatFileAssistant.ArchiveWorker", "bin", "Release"),
                Path.Combine(parent, "src", "CatFileAssistant.ArchiveWorker", "bin", "Debug") })
            {
                if (!Directory.Exists(folder)) continue;
                foreach (var relative in new[] { "net10.0-windows/win-x64/publish", "net10.0-windows/win-x64", "net10.0-windows" })
                {
                    var candidate = Path.Combine(folder, relative.Replace('/', Path.DirectorySeparatorChar), executable);
                    if (File.Exists(candidate)) return candidate;
                }
            }
        }
        return packaged;
    }

    private static IEnumerable<string> WorkspaceParents()
    {
        var directory = new DirectoryInfo(AppContext.BaseDirectory);
        for (var index = 0; directory is not null && index < 10; index++, directory = directory.Parent)
            if (File.Exists(Path.Combine(directory.FullName, "CatFileAssistant.sln"))) yield return directory.FullName;
    }
}

    public static bool IsSupportedArchive(string path)
    {
        var name = Path.GetFileName(path);
        return ExtensionsForExtraction.Any(extension => name.EndsWith(extension, StringComparison.OrdinalIgnoreCase))
            || SplitArchiveName().IsMatch(name) || OldRarVolumeName().IsMatch(name) || ZipVolumeName().IsMatch(name);
    }

    public static IReadOnlyList<string> NormalizeExtractionSources(IEnumerable<string> paths)
    {
        var result = new List<string>();
        var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        foreach (var source in paths)
        {
            var path = Path.GetFullPath(source);
            var directory = Path.GetDirectoryName(path)!;
            var name = Path.GetFileName(path);
            var split = SplitArchiveName().Match(name);
            var rar = PartRarVolumeName().Match(name);
            if (split.Success)
            {
                var digits = split.Groups["number"].Length;
                path = Path.Combine(directory, split.Groups["base"].Value + "." + "1".PadLeft(digits, '0'));
            }
            else if (rar.Success)
            {
                var root = rar.Groups["base"].Value;
                var digits = rar.Groups["number"].Length;
                path = Path.Combine(directory, root + ".part" + "1".PadLeft(digits, '0') + ".rar");
                if (!File.Exists(path) && Directory.Exists(directory))
                {
                    // Some native RAR sets use different padding widths for early and later volumes.
                    foreach (var sibling in Directory.EnumerateFiles(directory, "*.rar", SearchOption.TopDirectoryOnly))
                    {
                        var match = PartRarVolumeName().Match(Path.GetFileName(sibling));
                        if (match.Success && string.Equals(match.Groups["base"].Value, root, StringComparison.OrdinalIgnoreCase)
                            && long.TryParse(match.Groups["number"].Value, out var number) && number == 1)
                        { path = sibling; break; }
                    }
                }
            }
            else if (OldRarVolumeName().Match(name) is { Success: true } oldRar)
                path = Path.Combine(directory, oldRar.Groups["base"].Value + ".rar");
            else if (ZipVolumeName().Match(name) is { Success: true } zip)
                path = Path.Combine(directory, zip.Groups["base"].Value + ".zip");
            if (seen.Add(path)) result.Add(path);
        }
        return result.AsReadOnly();
    }

    private static ArchiveInputInspection Reject(string path, string code, string message, string suggestion)
        => new(path, false, 0, new(code, message, suggestion));
    [GeneratedRegex(@"^(?<base>.+\.(?:7z|zip|tar|tar\.gz|tar\.bz2|tar\.xz))\.(?<number>\d{3,})$", RegexOptions.IgnoreCase | RegexOptions.CultureInvariant)]
    private static partial Regex SplitArchiveName();
    [GeneratedRegex(@"^(?<base>.+)\.part(?<number>\d+)\.rar$", RegexOptions.IgnoreCase | RegexOptions.CultureInvariant)]
    private static partial Regex PartRarVolumeName();
    [GeneratedRegex(@"^(?<base>.+)\.r\d{2,}$", RegexOptions.IgnoreCase | RegexOptions.CultureInvariant)]
    private static partial Regex OldRarVolumeName();
    [GeneratedRegex(@"^(?<base>.+)\.z\d{2,}$", RegexOptions.IgnoreCase | RegexOptions.CultureInvariant)]
    private static partial Regex ZipVolumeName();
}

/// <summary>The worker receives secrets only through its stdin pipe, never command-line arguments.</summary>
internal static class ArchiveWorkerProcess
{
    private const int MaximumFrameCharacters = 256 * 1024;
    private static readonly JsonSerializerOptions JsonOptions = new(JsonSerializerDefaults.Web)
    { DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull };

    public static async Task<ArchiveWorkerOutcome> RunAsync(string workerPath, string dllPath,
        ArchiveWorkerRequest request, Action<ArchiveWorkerProgress>? progress,
        CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        var start = new ProcessStartInfo(Path.GetFullPath(workerPath))
        {
            UseShellExecute = false, CreateNoWindow = true,
            RedirectStandardInput = true, RedirectStandardOutput = true, RedirectStandardError = true,
            StandardInputEncoding = new UTF8Encoding(false),
            StandardOutputEncoding = Encoding.UTF8, StandardErrorEncoding = Encoding.UTF8,
            WorkingDirectory = Path.GetDirectoryName(Path.GetFullPath(workerPath))!
        };
        start.ArgumentList.Add("--dll=" + Path.GetFullPath(dllPath));
        using var process = new Process { StartInfo = start };
        using var job = new ArchiveWorkerJob();
        if (!process.Start()) throw new IOException(Locale.T("无法启动压缩包引擎。"));
        ArchiveWorkerResult? result = null;
        string? protocolError = null;
        long diagnosticCharacters = 0;
        var cancelled = false;
        Task? stdout = null, stderr = null;
        try
        {
            // The worker waits for its first request, so assignment precedes archive parsing.
            job.Assign(process);
            stdout = DrainLinesAsync(process.StandardOutput, line =>
            {
                if (line is null) { protocolError ??= "frame-too-large"; return; }
                if (string.IsNullOrWhiteSpace(line)) return;
                try
                {
                    using var document = JsonDocument.Parse(line);
                    var root = document.RootElement;
                    if (root.ValueKind != JsonValueKind.Object || !root.TryGetProperty("type", out var type)
                        || type.ValueKind != JsonValueKind.String)
                    { protocolError ??= "message-type"; return; }
                    if (type.GetString() == "progress")
                    {
                        var update = root.Deserialize<ArchiveWorkerProgress>(JsonOptions);
                        if (update is null || update.ProcessedBytes < 0 || update.ProcessedEntries < 0
                            || update.TotalBytes < 0 || update.TotalEntries < 0
                            || update.Percent is { } percent && !double.IsFinite(percent))
                        { protocolError ??= "progress-fields"; return; }
                        try { progress?.Invoke(update); } catch (Exception) { }
                    }
                    else if (type.GetString() == "result")
                    {
                        if (result is not null) { protocolError ??= "duplicate-result"; return; }
                        result = root.Deserialize<ArchiveWorkerResult>(JsonOptions);
                        if (result is null || result.SourceBytes < 0 || result.OutputBytes < 0 || result.EntryCount < 0
                            || result.Status is not ("completed" or "failed" or "cancelled"))
                        { result = null; protocolError ??= "result-fields"; }
                    }
                    else protocolError ??= "unknown-message";
                }
                catch (JsonException) { protocolError ??= "invalid-json"; }
            });
            stderr = DrainLinesAsync(process.StandardError, line =>
            {
                // Drain the pipe, but retain only a count: native diagnostics may contain secrets.
                diagnosticCharacters = Math.Min(long.MaxValue - MaximumFrameCharacters,
                    diagnosticCharacters) + (line?.Length ?? MaximumFrameCharacters);
            });
            var exit = process.WaitForExitAsync(CancellationToken.None);
            var cancelSignal = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
            using var registration = cancellationToken.Register(() => cancelSignal.TrySetResult());
            var initial = JsonSerializer.Serialize(request, JsonOptions);
            await process.StandardInput.WriteLineAsync(initial).WaitAsync(TimeSpan.FromSeconds(30), cancellationToken).ConfigureAwait(false);
            await process.StandardInput.FlushAsync().ConfigureAwait(false);
            if (await Task.WhenAny(exit, cancelSignal.Task).ConfigureAwait(false) != exit)
            {
                cancelled = true;
                await CancelAndWaitAsync(process, exit).ConfigureAwait(false);
            }
            else await exit.ConfigureAwait(false);
            await Task.WhenAll(stdout, stderr).WaitAsync(TimeSpan.FromSeconds(5)).ConfigureAwait(false);
            return new(process.ExitCode, result, cancelled || cancellationToken.IsCancellationRequested,
                protocolError, $"diagnostic-characters={diagnosticCharacters}");
        }
        catch
        {
            // No worker may outlive its request, including startup and protocol failures.
            try { if (!process.HasExited) process.Kill(entireProcessTree: true); }
            catch (InvalidOperationException) { }
            catch (Win32Exception) { }
            await process.WaitForExitAsync(CancellationToken.None).ConfigureAwait(false);
            if (stdout is not null && stderr is not null)
            {
                try { await Task.WhenAll(stdout, stderr).WaitAsync(TimeSpan.FromSeconds(5)).ConfigureAwait(false); }
                catch (IOException) { }
                catch (TimeoutException) { }
            }
            // A cleanup or pipe failure after a reported commit cannot delete or invalidate it.
            if (result?.Status == "completed")
                return new(process.ExitCode, result, cancellationToken.IsCancellationRequested,
                    protocolError ?? "worker-cleanup", $"diagnostic-characters={diagnosticCharacters}");
            throw;
        }
    }

    private static async Task CancelAndWaitAsync(Process process, Task exit)
    {
        var deadline = Stopwatch.StartNew();
        try
        {
            await process.StandardInput.WriteLineAsync("{\"action\":\"cancel\"}").WaitAsync(TimeSpan.FromSeconds(1)).ConfigureAwait(false);
            await process.StandardInput.FlushAsync().WaitAsync(TimeSpan.FromSeconds(1)).ConfigureAwait(false);
        }
        catch (IOException) { }
        catch (InvalidOperationException) { }
        catch (TimeoutException) { }
        var remaining = TimeSpan.FromSeconds(5) - deadline.Elapsed;
        try
        {
            if (remaining <= TimeSpan.Zero && !exit.IsCompleted) throw new TimeoutException();
            await exit.WaitAsync(remaining > TimeSpan.Zero ? remaining : TimeSpan.FromMilliseconds(1)).ConfigureAwait(false);
        }
        catch (TimeoutException)
        {
            try { if (!process.HasExited) process.Kill(entireProcessTree: true); }
            catch (InvalidOperationException) { }
            catch (Win32Exception) { }
            await exit.ConfigureAwait(false);
        }
    }

    private static async Task DrainLinesAsync(StreamReader reader, Action<string?> lineAction)
    {
        var buffer = new char[4096];
        var line = new StringBuilder();
        var tooLarge = false;
        while (true)
        {
            var read = await reader.ReadAsync(buffer.AsMemory()).ConfigureAwait(false);
            if (read == 0) break;
            for (var index = 0; index < read; index++)
            {
                var character = buffer[index];
                if (character == '\n')
                {
                    lineAction(tooLarge ? null : line.ToString().TrimEnd('\r'));
                    line.Clear(); tooLarge = false;
                }
                else if (!tooLarge)
                {
                    if (line.Length == MaximumFrameCharacters) { line.Clear(); tooLarge = true; }
                    else line.Append(character);
                }
            }
        }
        if (line.Length > 0 || tooLarge) lineAction(tooLarge ? null : line.ToString().TrimEnd('\r'));
    }

    internal static string Redact(string text, string? secret)
        => string.IsNullOrEmpty(secret) ? text : text.Replace(secret, "[redacted]", StringComparison.Ordinal);
}

/// <summary>Explicit, local output allowlist. A source extension never becomes an arbitrary encoder argument.</summary>
public static class CompressionCapabilities
{
    public static FormatPreset? ResolveSourcePreset(string path, ConversionMode mode, MediaInfo media)
    {
        var extension = Path.GetExtension(path).TrimStart('.').ToLowerInvariant();
        var codec = extension switch
        {
            "mp4" or "m4v" or "mov" or "mkv" => media.VideoCodec == "hevc" ? "hevc-kv" : "h264-open",
            "webm" => "vp9", "avi" => "mpeg4", "mp3" => "mp3", "m4a" or "aac" => "aac",
            "wav" => "wav", "flac" => "flac", "jpg" or "jpeg" => "jpg", "png" => "png",
            "webp" => "webp", "bmp" => "bmp", "gif" => "gif", _ => null
        };
        if (codec is null) return null;
        var container = extension switch
        {
            "mp4" or "m4v" or "m4a" => "mp4", "mov" => "mov", "mkv" => "matroska",
            "jpg" or "jpeg" or "png" or "webp" or "bmp" or "gif" => "image", _ => extension
        };
        var expected = codec switch
        {
            "h264-open" => "h264", "hevc-kv" => "hevc", "wav" => "pcm_s16le", "jpg" => "mjpeg", _ => codec
        };
        var preset = new FormatPreset("source-" + extension, extension.ToUpperInvariant() + Locale.T(" · 保留原格式"), extension, codec, mode, container, expected);
        return IsAllowed(preset) && IsCompatible(preset, media) ? preset : null;
    }

    public static IReadOnlyList<FormatPreset> Candidates(ConversionMode mode, MediaInfo media, IReadOnlyList<FormatPreset> presets)
        => presets.Concat(AdditionalPresets).Where(value => value.Mode == mode && IsAllowed(value) && IsCompatible(value, media))
            .DistinctBy(value => value.Id).ToArray();

    private static readonly FormatPreset[] AdditionalPresets =
    [
        new("compression-mov", Locale.T("MOV · 常见视频"), "mov", "h264-open", ConversionMode.Video, "mov", "h264"),
        new("compression-m4v", Locale.T("M4V · 常见视频"), "m4v", "h264-open", ConversionMode.Video, "mp4", "h264"),
        new("compression-aac", Locale.T("AAC · 原始音频"), "aac", "aac", ConversionMode.Audio, "aac", "aac"),
        new("compression-gif", Locale.T("GIF · 保留动画"), "gif", "gif", ConversionMode.Image, "image", "gif")
    ];

    public static bool IsCompatible(FormatPreset preset, MediaInfo media)
    {
        if (!IsAllowed(preset)) return false;
        if (preset.Mode == ConversionMode.Gif) return media.HasVideo;
        if (media.HasUnsupportedAnimation) return false;
        if (preset.Mode == ConversionMode.Image)
        {
            if (media.Animation?.IsAnimated == true && preset.Codec is not ("gif" or "webp")) return false;
            if (media.HasAlpha && preset.Codec is not ("png" or "webp" or "bmp" or "gif")) return false;
            // GIF only has binary transparency; never silently discard WebP/PNG partial alpha.
            if (media.HasAlpha && media.VideoCodec != "gif" && preset.Codec == "gif") return false;
        }
        var audios = media.AudioStreams?.Count ?? (media.HasAudio ? 1 : 0);
        if (preset.Mode is ConversionMode.Audio or ConversionMode.Extract && audios > 1 && preset.Extension != "m4a") return false;
        return (media.SubtitleStreams ?? []).All(value => CanCopySubtitle(preset, value.Codec));
    }

    public static bool CanCopySubtitle(FormatPreset preset, string codec) => preset.Extension switch
    {
        "mp4" or "m4v" or "mov" => codec == "mov_text",
        "mkv" => codec is "subrip" or "srt" or "ass" or "ssa" or "webvtt" or "hdmv_pgs_subtitle" or "dvd_subtitle" or "dvb_subtitle",
        "webm" => codec == "webvtt",
        _ => false
    };

    public static bool IsAllowed(FormatPreset preset) => (preset.Mode, preset.Codec, preset.Extension, preset.Container, preset.ExpectedCodec) switch

using System.Text;

namespace CatFileAssistant.Infrastructure;

/// <summary>A local bounded diagnostic log. Logging failures never invalidate conversion results.</summary>
public sealed class DiagnosticLog
{
    private const long MaximumFileBytes = 2 * 1024 * 1024;
    private const int MaximumEntryCharacters = 48 * 1024;
    private readonly object sync = new();
    private readonly string directory;
    public string FilePath => Path.Combine(directory, "diagnostics.log");

    public DiagnosticLog(string directory) => this.directory = Path.GetFullPath(directory);

    public string Write(string message, string? detail = null)
    {
        var id = Guid.NewGuid().ToString("N");
        var entry = $"[{DateTimeOffset.UtcNow:O}] [{id}] {message}{Environment.NewLine}{detail}{Environment.NewLine}";
        if (entry.Length > MaximumEntryCharacters) entry = entry[..MaximumEntryCharacters] + "\n[truncated]\n";
        try
        {
            lock (sync)
            {
                Directory.CreateDirectory(directory);
                var bytes = Encoding.UTF8.GetByteCount(entry);
                if (File.Exists(FilePath) && new FileInfo(FilePath).Length + bytes > MaximumFileBytes)
                    File.Move(FilePath, Path.Combine(directory, "diagnostics.previous.log"), true);
                File.AppendAllText(FilePath, entry, new UTF8Encoding(false));
            }
        }
        catch (IOException) { }
        catch (UnauthorizedAccessException) { }
        return id;
    }
}

    private async Task<ConversionResult> CompressAsync(ConversionRequest request, IProgress<ConversionProgress>? progress, CancellationToken cancellationToken)
    {

                if (copiedBytes > target) return Failure("INPUT_CHANGED", Locale.T("文件大小在处理时发生了变化。"), Locale.T("请重新添加这个文件再试。"), "Copied source exceeds the requested maximum.");

                var invalidCopy = ValidateCompressedContent(media, copied, preset, copied: true);
                if (invalidCopy is not null) return new(JobStatus.Failed, null, invalidCopy);
                var copyDecode = await ValidateDecodeAsync(temporary, copied, cancellationToken).ConfigureAwait(false);

                temporary = null;
                ReportSafely(progress, new(100, TimeSpan.FromSeconds(media.DurationSeconds), 0, MaximumCompressionAttempts));

                        }

                if (bytes > target)
                {
                    tuning.Reduce(target, bytes);
                    continue;
                }
                var produced = await ProbeAsync(temporary, cancellationToken).ConfigureAwait(false);
                var invalid = ValidateCompressedContent(media, produced, preset);
                if (invalid is not null) return new(JobStatus.Failed, null, invalid);
                var decoded = await ValidateDecodeAsync(temporary, produced, cancellationToken).ConfigureAwait(false);
                if (decoded.ExitCode != 0) return Failure("OUTPUT_DECODE", Locale.T("压缩文件没有通过完整读取检查。"), Locale.T("请尝试其他输出格式；原文件保持不变。"), decoded.StandardError);
                cancellationToken.ThrowIfCancellationRequested();
                statistics = new(media.SourceBytes, bytes, target, attempt);
                completed = CommitOutput(temporary, directory, Path.GetFileNameWithoutExtension(source) + "_压缩", preset.Extension);
                temporary = null;
                ReportSafely(progress, new(100, TimeSpan.FromSeconds(media.DurationSeconds), attempt, MaximumCompressionAttempts));
                return new(JobStatus.Completed, completed, null, statistics);
            }
            return Failure("TARGET_UNREACHABLE", Locale.T("已经尽力压缩，但这个大小暂时无法完整保存文件。"),
                Locale.F($"请把最大大小调大一些，或选择更高效的格式。本次最小结果约为 {smallest / 1_000_000d:0.###} MB。"),
                $"Maximum attempts reached. Target={target}, smallest={smallest}");
        }
        catch (OperationCanceledException) { return completed is null ? new(JobStatus.Cancelled, null, null) : new(JobStatus.Completed, completed, null, statistics); }
        catch (ConversionEngineException exception) { return new(JobStatus.Failed, null, exception.Error); }
        catch (UnauthorizedAccessException exception) { return ResultForException(completed, "ACCESS_DENIED", Locale.T("没有权限读取文件或保存结果。"), Locale.T("请选择有写入权限的文件夹。"), exception); }
        catch (IOException exception) { return ResultForException(completed, "FILE_IO", Locale.T("文件读取或保存遇到了问题。"), Locale.T("请检查文件和保存位置，然后重试。"), exception); }
        catch (Win32Exception exception) { return ResultForException(completed, "ENGINE_START", Locale.T("压缩引擎无法启动。"), Locale.T("请完整解压软件包。"), exception); }
        catch (Exception exception) { return ResultForException(completed, "COMPRESSION_UNEXPECTED", Locale.T("这个文件暂时无法安全压缩。"), Locale.T("请调大大小或更换兼容格式。"), exception); }
        finally
        {
            if (temporary is not null)
            {
                try { File.Delete(temporary); }
                catch (IOException exception) { log?.Write("Compression temporary cleanup failed.", exception.ToString()); }
                catch (UnauthorizedAccessException exception) { log?.Write("Compression temporary cleanup denied.", exception.ToString()); }
            }
        }
    }

    private ConversionError? ValidateCompressedContent(MediaInfo source, MediaInfo output, FormatPreset preset, bool copied = false)
    {
        string? problem = null;
        if (copied ? source.FormatName != output.FormatName || source.VideoCodec != output.VideoCodec || source.AudioCodec != output.AudioCodec
            : !MatchesPreset(output, preset)) problem = "Container or codec differs from requested preset.";
        else if ((source.AudioStreams?.Count ?? 0) != (output.AudioStreams?.Count ?? 0)) problem = "Audio stream count changed.";
        else if ((source.SubtitleStreams?.Count ?? 0) != (output.SubtitleStreams?.Count ?? 0)) problem = "Subtitle stream count changed.";
        else if (source.HasAlpha && !output.HasAlpha) problem = "Alpha support disappeared.";
        else if (source.Animation?.IsAnimated == true)
        {
            var animation = output.Animation;
            var sourceAnimation = source.Animation;
            var expectedLoop = preset.Codec == "gif" ? AnimationMetadata.GifLoop(sourceAnimation)
                : sourceAnimation.IsGif ? sourceAnimation.LoopCount switch { -1 => 1, 0 => 0, _ => Math.Min(65535, sourceAnimation.LoopCount + 1) } : sourceAnimation.LoopCount;
            if (animation?.IsAnimated != true || animation.FrameCount != sourceAnimation.FrameCount) problem = "Animation frames disappeared.";
            else if (Math.Abs(animation.DurationMilliseconds - sourceAnimation.DurationMilliseconds) > (preset.Codec == "gif" ? 10 : 1)) problem = "Animation duration changed.";
            else if (animation.LoopCount != expectedLoop) problem = "Animation loop changed.";
        }
        else if (preset.Mode != ConversionMode.Image)
        {
            var tolerance = Math.Max(.15, source.FrameRate > 0 ? 2 / source.FrameRate : .15);
            if (Math.Abs(output.DurationSeconds - source.DurationSeconds) > tolerance) problem = $"Duration changed: {source.DurationSeconds} => {output.DurationSeconds}.";
            if (problem is null && source.AudioStreams is { } audios && output.AudioStreams is { } outputs)
            {
                for (var i = 0; i < audios.Count; i++)
                    if (audios[i].DurationSeconds > 0 && outputs[i].DurationSeconds > 0 && Math.Abs(audios[i].DurationSeconds - outputs[i].DurationSeconds) > .15)
                        { problem = "An audio track was truncated."; break; }
            }
        }
        return problem is null ? null : ErrorFor("OUTPUT_CONTENT", Locale.T("这个压缩结果没有完整保留文件内容。"), Locale.T("请调大目标大小，或选择其他兼容格式；原文件保持不变。"), problem);
    }

    private static async Task CopyAsync(string source, string temporary, CancellationToken cancellationToken)
    {
        using var input = new FileStream(source, FileMode.Open, FileAccess.Read, FileShare.Read, 81920, true);
        using var output = new FileStream(temporary, FileMode.Truncate, FileAccess.Write, FileShare.None, 81920, true);
        await input.CopyToAsync(output, cancellationToken).ConfigureAwait(false);

    }

    private sealed class CompressionTuning
    {
        public double Scale { get; private set; } = 1;
        public int Quality { get; private set; } = 85;
        public int JpegQuality { get; private set; } = 2;
        public int Colors { get; private set; } = 256;
        public int VideoBitRate { get; private set; }
        public int AudioBitRate { get; private set; }
        public int AudioSampleRate { get; private set; } = 48000;
        public int AudioChannels { get; private set; } = 2;
        public double FramesPerSecond { get; private set; }
        private readonly MediaInfo media;
        private readonly FormatPreset preset;
        private readonly double minimumScale;

        public CompressionTuning(MediaInfo media, FormatPreset preset, long target)
        {
            this.media = media; this.preset = preset;
            var shortest = Math.Max(1, Math.Min(media.Width, media.Height));
            minimumScale = Math.Min(1, (preset.Mode == ConversionMode.Video ? 144d : 16d) / shortest);
            FramesPerSecond = media.FrameRate > 0 ? Math.Min(media.FrameRate, 30) : 30;
            var count = Math.Max(1, media.AudioStreams?.Count ?? (media.HasAudio ? 1 : 0));
            var totalRate = Math.Max(1, target * 8d * .90 / Math.Max(.001, media.DurationSeconds));
            AudioBitRate = (int)Math.Clamp(preset.Mode == ConversionMode.Video ? totalRate * .18 / count : totalRate / count, 8000, 320000);
            if (preset.Mode == ConversionMode.Video) AudioBitRate = Math.Min(128000, Math.Max(16000, AudioBitRate));
            VideoBitRate = (int)Math.Clamp(totalRate - (media.HasAudio ? AudioBitRate * count : 0), 1000, 100_000_000);
            if (preset.Mode == ConversionMode.Video)
            {
                var estimatedPixels = VideoBitRate / Math.Max(1, FramesPerSecond * .075);
                Scale = Math.Clamp(Math.Sqrt(estimatedPixels / Math.Max(1, (double)media.Width * media.Height)), minimumScale, 1);
            }
            if (preset.Codec == "wav")
                AudioSampleRate = (int)Math.Clamp(totalRate / Math.Max(1, (media.AudioStreams?.FirstOrDefault()?.Channels ?? 2) * 16), 8000, 48000);
            SelectAudioLayout();
        }

        public void Reduce(long target, long actual)
        {
            var ratio = Math.Clamp((double)target / actual * .85, .05, .88);
            if (preset.Mode == ConversionMode.Image)
            {
                if (preset.Codec == "jpg" && JpegQuality < 28) { JpegQuality = Math.Min(28, JpegQuality + 5); if (JpegQuality < 17) return; }
                if (preset.Codec == "webp" && Quality > 15) { Quality = Math.Max(15, (int)(Quality * ratio)); if (Quality > 35) return; }
                if (preset.Codec == "gif" && Colors > 8) Colors = Math.Max(8, Colors / 2);
                Scale = Math.Max(minimumScale, Scale * Math.Sqrt(ratio));
            }
            else
            {
                VideoBitRate = Math.Max(1000, (int)(VideoBitRate * ratio));
                AudioBitRate = Math.Max(8000, (int)(AudioBitRate * ratio));
                if (preset.Mode == ConversionMode.Video)
                {
                    Scale = Math.Max(minimumScale, Scale * Math.Sqrt(ratio));
                    FramesPerSecond = Math.Max(Math.Min(media.FrameRate > 0 ? media.FrameRate : 12, 12), FramesPerSecond * .85);
                }
                if (preset.Codec is "wav" or "flac") AudioSampleRate = Math.Max(8000, (int)(AudioSampleRate * ratio));
                SelectAudioLayout();
            }
        }

        private void SelectAudioLayout()
        {
            if (AudioBitRate < 48000) { AudioChannels = 1; AudioSampleRate = Math.Min(AudioSampleRate, AudioBitRate < 24000 ? 16000 : 22050); }
            else if (AudioBitRate < 96000) AudioSampleRate = Math.Min(AudioSampleRate, 32000);
        }
    }

    private static IEnumerable<string> BuildCompressionArguments(string source, string temporary, FormatPreset preset, MediaInfo media, CompressionTuning tuning)
    {
        var args = new List<string> { "-nostdin", "-hide_banner", "-loglevel", "error", "-progress", "pipe:1", "-nostats" };
        AddAnimationInputOptions(args, media);
        args.AddRange(["-i", source, "-map_metadata", "0", "-map_chapters", "0"]);
        // iw/ih refer to decoded, automatically rotated frames, unlike probe's coded dimensions.
        var factor = tuning.Scale.ToString("R", CultureInfo.InvariantCulture);
        var scale = $"scale=w='max(1,floor(iw*{factor}))':h='max(1,floor(ih*{factor}))':flags=lanczos";
        // Expand indexed palettes before resizing. A pal8 -> pal8 scale can silently drop PNG tRNS alpha.
        if (preset.Mode == ConversionMode.Image && media.HasAlpha) scale = "format=rgba," + scale;
        if (preset.Mode == ConversionMode.Video)
        {
            args.AddRange(["-map", "0:V:0", "-map", "0:a?", "-map", "0:s?", "-c:s", "copy", "-dn", "-pix_fmt", "yuv420p",
                "-vf", scale + ",pad=ceil(iw/2)*2:ceil(ih/2)*2", "-r", tuning.FramesPerSecond.ToString("0.###", CultureInfo.InvariantCulture)]);
            switch (preset.Codec)
            {
                case "h264-open": args.AddRange(["-c:v", "libopenh264", "-rc_mode", "bitrate", "-b:v", tuning.VideoBitRate.ToString(), "-maxrate", tuning.VideoBitRate.ToString(), "-allow_skip_frames", "1", "-profile:v", "main"]); break;
                case "hevc-kv": args.AddRange(["-c:v", "libkvazaar", "-b:v", tuning.VideoBitRate.ToString(), "-kvazaar-params", "preset=fast"]); break;
                case "vp9": args.AddRange(["-c:v", "libvpx-vp9", "-b:v", tuning.VideoBitRate.ToString(), "-deadline", "good", "-cpu-used", "4"]); break;
                case "mpeg4": args.AddRange(["-c:v", "mpeg4", "-b:v", tuning.VideoBitRate.ToString()]); break;
            }
            AddCompressedAudio(args, preset.Codec == "vp9" ? "opus" : preset.Codec == "mpeg4" ? "mp3" : "aac", media, tuning);
            if (preset.Extension is "mp4" or "m4v" or "mov")
            {
                args.AddRange(["-f", preset.Extension == "mov" ? "mov" : "mp4", "-movflags", "+faststart"]);
                if (preset.Codec == "hevc-kv") args.AddRange(["-tag:v", "hvc1"]);
            }
        }
        else if (preset.Mode is ConversionMode.Audio or ConversionMode.Extract)
        {
            args.AddRange(["-map", "0:a", "-map", "0:s?", "-c:s", "copy", "-vn", "-dn"]);
            AddCompressedAudio(args, preset.Codec, media, tuning);
            if (preset.Extension == "aac") args.AddRange(["-f", "adts"]);
        }
        else
        {
            args.AddRange(["-map", "0:v:0", "-an", "-sn", "-dn"]);
            if (media.Animation?.IsAnimated != true) args.AddRange(["-frames:v", "1"]);
            if (preset.Codec == "gif")
            {
                args.AddRange(["-filter_complex", $"[0:v]{scale},split[frames][palette];[palette]palettegen=max_colors={tuning.Colors}:reserve_transparent=1[pal];[frames][pal]paletteuse=dither=bayer:bayer_scale=3:alpha_threshold=128",
                    "-fps_mode", "passthrough", "-enc_time_base", "1/100", "-loop", (media.Animation is { } animation ? AnimationMetadata.GifLoop(animation) : -1).ToString()]);
                // A filtered stream must be mapped, rather than also encoding the original unfiltered stream.
                var mapping = args.IndexOf("-map"); args.RemoveRange(mapping, 2);
                if (media.Animation is { } timed) args.AddRange(["-final_delay", Math.Max(1, timed.FrameDurationsMilliseconds[^1] / 10).ToString()]);
                args.AddRange(["-c:v", "gif"]);
            }
            else
            {
                args.AddRange(["-vf", scale]);
                args.AddRange(preset.Codec switch
                {
                    "jpg" => ["-c:v", "mjpeg", "-q:v", tuning.JpegQuality.ToString()],
                    "png" => media.HasAlpha
                        ? ["-c:v", "png", "-pix_fmt", "rgba", "-compression_level", "9", "-pred", "mixed"]
                        : ["-c:v", "png", "-compression_level", "9", "-pred", "mixed"],
                    "bmp" => ["-c:v", "bmp", "-pix_fmt", media.HasAlpha ? "bgra" : "bgr24"],
                    "webp" => ["-c:v", "libwebp", "-quality", tuning.Quality.ToString(), "-compression_level", "6"],
                    _ => throw new InvalidDataException("Unknown compression image format.")
                });
                if (media.Animation?.IsAnimated == true) args.AddRange(["-fps_mode", "passthrough", "-enc_time_base", "1/1000", "-loop", "0"]);
            }
        }
        args.AddRange(["-y", temporary]); return args;
    }

    private static void AddCompressedAudio(List<string> args, string codec, MediaInfo media, CompressionTuning tuning)
    {
        args.AddRange(codec switch
        {
            "mp3" => ["-c:a", "libmp3lame", "-b:a", Mp3Rate(tuning.AudioBitRate).ToString()],
            "aac" => ["-c:a", "aac", "-b:a", tuning.AudioBitRate.ToString()],
            "opus" => ["-c:a", "libopus", "-b:a", tuning.AudioBitRate.ToString()],
            "wav" => ["-c:a", "pcm_s16le"],
            "flac" => ["-c:a", "flac", "-sample_fmt", "s16", "-compression_level", "12"],
            _ => throw new InvalidDataException("Unknown compression audio format.")
        });
        var streams = media.AudioStreams ?? [];
        for (var index = 0; index < streams.Count; index++)
        {
            var stream = streams[index];
            var rate = Math.Min(stream.SampleRate > 0 ? stream.SampleRate : 48000, tuning.AudioSampleRate);
            if (codec == "opus") rate = rate >= 48000 ? 48000 : rate >= 24000 ? 24000 : rate >= 16000 ? 16000 : rate >= 12000 ? 12000 : 8000;
            args.AddRange([$"-ar:a:{index}", Math.Max(8000, rate).ToString(), $"-ac:a:{index}", Math.Min(Math.Max(1, stream.Channels), tuning.AudioChannels).ToString()]);
            if (!string.IsNullOrWhiteSpace(stream.Language)) args.AddRange([$"-metadata:s:a:{index}", "language=" + stream.Language]);
            if (!string.IsNullOrWhiteSpace(stream.Title)) args.AddRange([$"-metadata:s:a:{index}", "title=" + stream.Title]);
        }
    }
    private static int Mp3Rate(int requested) => new[] { 8000, 16000, 24000, 32000, 40000, 48000, 56000, 64000, 80000, 96000, 112000, 128000, 160000, 192000, 224000, 256000, 320000 }.Last(value => value <= Math.Max(8000, requested));

    {
        if (media.VideoCodec is "gif" or "webp_anim") args.AddRange(["-ignore_loop", "1", "-min_delay", "1"]);
    }

    private static bool HasUnsupportedImageSequence(string path, string format)
    {
        if (format == "apng") return true;
        var extension = Path.GetExtension(path).ToLowerInvariant();
        using var file = File.OpenRead(path);
        if (extension == ".png" && file.Length >= 8)
        {
            file.Position = 8; Span<byte> chunk = stackalloc byte[8];
            while (file.Position + 8 <= file.Length)
            {
                file.ReadExactly(chunk); var length = BinaryPrimitives.ReadUInt32BigEndian(chunk[..4]);
                if (chunk[4..].SequenceEqual("acTL"u8)) return true;
                if (chunk[4..].SequenceEqual("IDAT"u8) || chunk[4..].SequenceEqual("IEND"u8)) break;
                if (file.Position + length + 4 > file.Length) break;
                file.Position += length + 4;
            }
        }
        if (extension is ".tif" or ".tiff" && file.Length >= 8)
        {
            Span<byte> header = stackalloc byte[8]; file.ReadExactly(header);
            var little = header[0] == 'I' && header[1] == 'I';
            var big = header[0] == 'M' && header[1] == 'M';
            if (!little && !big) return true;
            var version = little ? BinaryPrimitives.ReadUInt16LittleEndian(header[2..4]) : BinaryPrimitives.ReadUInt16BigEndian(header[2..4]);
            if (version != 42) return true; // BigTIFF needs a distinct parser; do not silently lose pages.
            var offset = little ? BinaryPrimitives.ReadUInt32LittleEndian(header[4..]) : BinaryPrimitives.ReadUInt32BigEndian(header[4..]);
            if (offset + 2L > file.Length) return true;
            file.Position = offset; Span<byte> countBytes = stackalloc byte[2]; file.ReadExactly(countBytes);
            var count = little ? BinaryPrimitives.ReadUInt16LittleEndian(countBytes) : BinaryPrimitives.ReadUInt16BigEndian(countBytes);
            var next = offset + 2L + count * 12L;
            if (next + 4 > file.Length) return true;
            file.Position = next; Span<byte> nextBytes = stackalloc byte[4]; file.ReadExactly(nextBytes);
            return (little ? BinaryPrimitives.ReadUInt32LittleEndian(nextBytes) : BinaryPrimitives.ReadUInt32BigEndian(nextBytes)) != 0;
        }
        return false;
    }
}

    private static IEnumerable<string> BuildArguments(string source, string temporary, FormatPreset preset)
    {
        var args = new List<string> { "-nostdin", "-hide_banner", "-loglevel", "error", "-progress", "pipe:1", "-nostats", "-i", source };
        if (preset.Mode == ConversionMode.Video)
        {
            args.AddRange(["-map", "0:V:0", "-map", "0:a:0?", "-sn", "-dn"]);
            switch (preset.Codec)
            {
                case "h264-open": args.AddRange(["-c:v", "libopenh264", "-b:v", "4M", "-pix_fmt", "yuv420p", "-c:a", "aac", "-b:a", "192k"]); break;
                case "hevc-kv": args.AddRange(["-c:v", "libkvazaar", "-b:v", "3M", "-pix_fmt", "yuv420p", "-c:a", "aac", "-b:a", "160k"]); break;
                case "vp9": args.AddRange(["-c:v", "libvpx-vp9", "-crf", "32", "-b:v", "0", "-c:a", "libopus", "-b:a", "128k"]); break;
                case "mpeg4": args.AddRange(["-c:v", "mpeg4", "-q:v", "4", "-c:a", "libmp3lame", "-b:a", "192k"]); break;
            }
            // These encoders require even chroma dimensions, including odd-size source video.
            args.AddRange(["-vf", "pad=ceil(iw/2)*2:ceil(ih/2)*2"]);
            if (preset.Extension == "mp4")
            {
                if (preset.Codec == "hevc-kv") args.AddRange(["-tag:v", "hvc1"]);
                args.AddRange(["-movflags", "+faststart"]);
            }
        }
        else if (preset.Mode is ConversionMode.Audio or ConversionMode.Extract)
        {
            args.AddRange(["-map", "0:a:0", "-vn", "-sn", "-dn"]);
            args.AddRange(preset.Codec switch
            {
                "mp3" => ["-c:a", "libmp3lame", "-q:a", "2"],
                "aac" => ["-c:a", "aac", "-b:a", "192k"],
                "wav" => ["-c:a", "pcm_s16le"],
                "flac" => ["-c:a", "flac"],
                _ => throw new InvalidDataException(Locale.T("未知音频预设。"))
            });
        }
        else
        {
            args.AddRange(["-map", "0:v:0", "-frames:v", "1", "-an", "-sn", "-dn"]);
            args.AddRange(preset.Codec switch
            {
                "jpg" => ["-c:v", "mjpeg", "-q:v", "2"],
                "png" => ["-c:v", "png"],
                "webp" => ["-c:v", "libwebp", "-quality", "85"],
                "bmp" => ["-c:v", "bmp"],
                _ => throw new InvalidDataException(Locale.T("未知图片预设。"))
            });
        }
        args.AddRange(["-y", temporary]);
        return args;

                return GifTimingFailure("The final selected frame would exceed GIF's maximum delay.");

                        && long.TryParse(line.AsSpan(12), NumberStyles.Integer, CultureInfo.InvariantCulture, out var microseconds))
                    {
                        var seconds = Math.Max(0, microseconds / 1_000_000d);
                        ReportSafely(progress, new(Math.Clamp(seconds / timeline.DurationSeconds * 95, 0, 95), TimeSpan.FromSeconds(seconds)));
                    }
                }, cancellationToken).ConfigureAwait(false);

                    || outcome.StandardError.Contains("Invalid, non monotonically", StringComparison.OrdinalIgnoreCase)
                    ? GifTimingFailure(outcome.StandardError)
                    : new(JobStatus.Failed, null, ErrorFromEngine(outcome.StandardError, outcome.ExitCode));
            if (new FileInfo(temporary).Length == 0)
                return Failure("OUTPUT_EMPTY", Locale.T("没有生成完整的 GIF。"), Locale.T("请确认视频可以正常播放，然后重试。"), outcome.StandardError);
            ReportSafely(progress, new(null, TimeSpan.FromSeconds(timeline.DurationSeconds)));
            var raw = AnimationMetadata.ReadGifRaw(temporary);
            var animation = raw.Animation;
            var (expectedWidth, expectedHeight) = GifDimensions(timeline, profile);
            var expectedAspect = profile.MaximumEdge == 0 ? timeline.SampleAspectRatio : 1;

        catch (ConversionEngineException exception)
        { return completed is null ? new(JobStatus.Failed, null, exception.Error) : new(JobStatus.Completed, completed, null, GifStatistics: statistics); }
        catch (UnauthorizedAccessException exception)
        { return ResultForException(completed, "ACCESS_DENIED", Locale.T("没有权限读取视频或保存 GIF。"), Locale.T("请选择有读写权限的文件夹。"), exception); }
        catch (IOException exception)
        { return ResultForException(completed, "FILE_IO", Locale.T("视频读取或 GIF 保存遇到了问题。"), Locale.T("请检查磁盘空间、文件占用和保存位置，然后重试。"), exception); }
        catch (Win32Exception exception)
        { return ResultForException(completed, "ENGINE_START", Locale.T("GIF 转换引擎无法启动。"), Locale.T("请完整解压软件包。"), exception); }

        {

            {

                catch (IOException exception) { log?.Write("GIF temporary cleanup failed.", exception.ToString()); }

        Locale.T("这个视频暂时无法在当前档位完整保留帧数和播放时长。"), Locale.T("请改用高档或更低的档位；原视频保持不变。"), detail);

    private async Task<GifTimeline> ReadGifTimelineAsync(string source, GifProfile profile,
        IProgress<ConversionProgress>? progress, CancellationToken cancellationToken)
    {
        var frames = 0; var selected = 0; var width = 0; var height = 0; var rotation = 0d; var aspect = 1d;
        var first = double.NaN; var previous = double.NaN; var last = 0d; var lastDuration = 0d;
        var lastSelected = double.NaN; var lastGap = 0d; var streamDuration = 0d;
        string? invalid = null;
        var outcome = await ProcessRunner.RunAsync(Path.Combine(engineDirectory, "ffprobe.exe"),
            ["-v", "error", "-select_streams", "V:0", "-show_frames", "-show_streams", "-show_entries",
                "frame=best_effort_timestamp_time,duration_time,width,height:stream=index,width,height,duration,sample_aspect_ratio:stream_side_data=rotation",
                "-of", "compact=p=0", source], line =>
            {
                // Numeric compact fields are streamed: long videos never build a frame-sized JSON document.
                var fields = new Dictionary<string, string>(StringComparer.Ordinal);
                foreach (var part in line.Split('|'))
                {
                    var pair = part.Split('=', 2);
                    if (pair.Length == 2)
                        fields[pair[0].EndsWith(":rotation", StringComparison.Ordinal) ? "rotation" : pair[0]] = pair[1];
                }
                if (fields.TryGetValue("rotation", out var angle) && double.TryParse(angle, CultureInfo.InvariantCulture, out var degrees)) rotation = degrees;
                if (fields.ContainsKey("index"))
                {
                    if (fields.TryGetValue("duration", out var duration)) double.TryParse(duration, CultureInfo.InvariantCulture, out streamDuration);
                    if (fields.TryGetValue("sample_aspect_ratio", out var ratio))
                    {
                        var terms = ratio.Split(':');
                        if (terms.Length == 2 && double.TryParse(terms[0], CultureInfo.InvariantCulture, out var numerator)
                            && double.TryParse(terms[1], CultureInfo.InvariantCulture, out var denominator) && numerator > 0 && denominator > 0)
                            aspect = numerator / denominator;
                    }
                    return;
                }
                if (!fields.ContainsKey("width")) return;
                if (!fields.TryGetValue("best_effort_timestamp_time", out var value)
                    || !double.TryParse(value, CultureInfo.InvariantCulture, out var timestamp) || !double.IsFinite(timestamp))
                { invalid = "A decoded video frame has no finite presentation timestamp."; return; }
                if (!double.IsNaN(previous) && timestamp <= previous) invalid = "Source video timestamps are not strictly increasing.";
                if (double.IsNaN(first)) first = timestamp;
                var position = timestamp - first;
                if (!double.IsNaN(previous)) lastGap = timestamp - previous;
                previous = timestamp; last = position;
                lastDuration = fields.TryGetValue("duration_time", out var time)
                    && double.TryParse(time, CultureInfo.InvariantCulture, out var durationSeconds) && double.IsFinite(durationSeconds) && durationSeconds > 0 ? durationSeconds : 0;
                if (frames == 0)
                {
                    int.TryParse(fields.GetValueOrDefault("width"), out width);
                    int.TryParse(fields.GetValueOrDefault("height"), out height);
                }
                frames++;
                if (double.IsNaN(lastSelected) || profile.MaximumFps == 0
                    || Math.Floor(position * profile.MaximumFps + 1e-4) > Math.Floor(lastSelected * profile.MaximumFps + 1e-4))
                { selected++; lastSelected = position; }
                if (frames % 250 == 0) ReportSafely(progress, new(null, TimeSpan.FromSeconds(Math.Max(0, position))));
            }, cancellationToken).ConfigureAwait(false);
        if (outcome.ExitCode != 0 || !string.IsNullOrWhiteSpace(outcome.StandardError))
            throw new ConversionEngineException(ErrorFor("GIF_SOURCE_DECODE", Locale.T("这个视频没有通过完整读取检查。"), Locale.T("请确认视频文件完整并可以正常播放。"), outcome.StandardError));
        if (invalid is not null || frames == 0 || width <= 0 || height <= 0)
            throw new ConversionEngineException(ErrorFor("GIF_SOURCE_TIMING", Locale.T("暂时无法读取这个视频的完整画面和时间信息。"), Locale.T("请确认视频可以正常播放，或换一个视频。"), invalid ?? "No usable video frames or dimensions."));
        var end = last + (lastDuration > 0 ? lastDuration : lastGap > 0 ? lastGap : streamDuration);
        if (lastDuration <= 0 && streamDuration > last) end = streamDuration;
        if (!double.IsFinite(end) || end <= last || end <= 0)
            throw new ConversionEngineException(ErrorFor("GIF_SOURCE_TIMING", Locale.T("暂时无法读取这个视频的完整时长。"), Locale.T("请换一个具有完整时间信息的视频。"), "The last decoded frame has no valid duration."));
        if (Math.Abs(Math.Abs(rotation) % 180 - 90) < .01) { (width, height) = (height, width); aspect = 1 / aspect; }
        return new(frames, selected, end, lastSelected, width, height, aspect);
    }

    private static (int Width, int Height) GifDimensions(GifTimeline timeline, GifProfile profile)
    {
        if (profile.MaximumEdge == 0) return (timeline.Width, timeline.Height);
        // Correct non-square pixels by reducing one axis, never enlarging decoded source pixels.
        var width = timeline.Width * Math.Min(1, timeline.SampleAspectRatio);
        var height = timeline.Height * Math.Min(1, 1 / timeline.SampleAspectRatio);
        var factor = Math.Min(1, profile.MaximumEdge / Math.Max(width, height));
        return (Math.Max(1, (int)Math.Floor(width * factor)), Math.Max(1, (int)Math.Floor(height * factor)));
    }

    private static IEnumerable<string> BuildGifArguments(string source, string temporary, GifProfile profile, GifTimeline timeline, int finalDelay)
    {
        var filters = "setpts=PTS-STARTPTS";
        if (profile.MaximumFps > 0)
            // select only removes actual source frames; unlike fps, it never duplicates sparse frames.
            filters += $",select='isnan(prev_selected_t)+gt(floor(t*{profile.MaximumFps}+0.0001),floor(prev_selected_t*{profile.MaximumFps}+0.0001))'";
        if (profile.MaximumEdge > 0)
        {
            var (width, height) = GifDimensions(timeline, profile);
            filters += $",scale={width}:{height}:flags=lanczos,setsar=1";
        }
        var graph = $"[0:V:0]{filters},split[frames][palette];[palette]palettegen=max_colors={profile.Colors}:reserve_transparent=1:stats_mode=single[pal];[frames][pal]paletteuse=new=1:dither={profile.Dither}[gif]";
        return ["-nostdin", "-hide_banner", "-loglevel", "error", "-xerror", "-progress", "pipe:1", "-nostats", "-i", source,
            "-filter_complex", graph, "-map", "[gif]", "-an", "-sn", "-dn", "-c:v", "gif", "-fps_mode", "passthrough",
            "-enc_time_base", "1/100", "-loop", "0", "-final_delay", finalDelay.ToString(CultureInfo.InvariantCulture), "-f", "gif", "-y", temporary];
    }
}

using System.Collections;

namespace CatFileAssistant.Infrastructure;

internal static class GilbertCurve
{
    public static int[] CreatePath(int width, int height, CancellationToken cancellationToken)
    {
        var path = new int[checked(width * height)];
        var seen = new BitArray(path.Length);
        var position = 0;
        void Add(int x, int y)
        {
            if ((position & 4095) == 0) cancellationToken.ThrowIfCancellationRequested();
            if ((uint)x >= width || (uint)y >= height || position >= path.Length)
                throw new InvalidDataException("Gilbert path is outside the image.");
            var index = y * width + x;
            if (seen[index]) throw new InvalidDataException("Gilbert path contains duplicate pixels.");
            seen[index] = true;
            path[position++] = index;
        }

        void Generate(int x, int y, int ax, int ay, int bx, int by)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var w = Math.Abs(ax + ay); var h = Math.Abs(bx + by);
            var dax = Math.Sign(ax); var day = Math.Sign(ay);
            var dbx = Math.Sign(bx); var dby = Math.Sign(by);
            if (h == 1)
            {
                for (var index = 0; index < w; index++) { Add(x, y); x += dax; y += day; }
                return;
            }
            if (w == 1)
            {
                for (var index = 0; index < h; index++) { Add(x, y); x += dbx; y += dby; }
                return;
            }
            var ax2 = FloorHalf(ax); var ay2 = FloorHalf(ay);
            var bx2 = FloorHalf(bx); var by2 = FloorHalf(by);
            var w2 = Math.Abs(ax2 + ay2); var h2 = Math.Abs(bx2 + by2);
            if (2 * w > 3 * h)
            {
                if (w2 % 2 != 0 && w > 2) { ax2 += dax; ay2 += day; }
                Generate(x, y, ax2, ay2, bx, by);
                Generate(x + ax2, y + ay2, ax - ax2, ay - ay2, bx, by);
            }
            else
            {
                if (h2 % 2 != 0 && h > 2) { bx2 += dbx; by2 += dby; }
                Generate(x, y, bx2, by2, ax2, ay2);
                Generate(x + bx2, y + by2, ax, ay, bx - bx2, by - by2);
                Generate(x + (ax - dax) + (bx2 - dbx), y + (ay - day) + (by2 - dby),
                    -bx2, -by2, -(ax - ax2), -(ay - ay2));
            }
        }

        if (width >= height) Generate(0, 0, width, 0, 0, height);
        else Generate(0, 0, 0, height, width, 0);
        if (position != path.Length) throw new InvalidDataException("Gilbert path did not cover the image.");
        return path;
    }

    // Python's // rounds toward negative infinity; C# integer division truncates toward zero.
    private static int FloorHalf(int value) => value >= 0 ? value / 2 : (value - 1) / 2;
}

    public async Task<ImageObfuscationResult> ProcessAsync(ImageObfuscationRequest request,

        string? snapshotPath = null; FileStream? snapshot = null;

            (snapshotPath, snapshot) = CreateSourceSnapshot(sourcePath, request.Id);
            await CopySourceSnapshotAsync(source, snapshot, token).ConfigureAwait(false);
            var inspected = await InspectSourceAsync(snapshotPath, snapshot, token).ConfigureAwait(false);
            if (request.Operation == ImageObfuscationOperation.Obfuscate && inspected.Marker is not null)
                throw Failure("already_marked", Locale.T("这张图片已有混淆标记。"), Locale.T("请先解混淆；需要多次处理时，请直接设置混淆次数。"));

            var iterations = inspected.Marker?.Iterations ?? request.Options.Iterations;
            var directory = Path.GetFullPath(request.OutputDirectory); Directory.CreateDirectory(directory);
            normalizedPath = ReserveTemporary(directory, "normalized", request.Id);
            temporary = ReserveTemporary(directory, "output", request.Id);
            Report(progress, 5, Locale.T("读取像素"));
            await NormalizeAsync(snapshotPath, normalizedPath, inspected.BitDepth, token).ConfigureAwait(false);
            var normalized = ReadPng(normalizedPath, token);
            ValidateDimensions(normalized.Width, normalized.Height);
            var width = normalized.Width; var height = normalized.Height; var bitDepth = normalized.BitDepth;

            var originalDigest = Convert.ToHexStringLower(SHA256.HashData(pixels));
            Report(progress, 15, Locale.T("重排像素"));

                mode, iterations, request.Operation == ImageObfuscationOperation.Deobfuscate,
                fraction => Report(progress, 15 + fraction * 55, Locale.T("重排像素")), token), token).ConfigureAwait(false);

                if (width != stored.Width || height != stored.Height || bitDepth != stored.BitDepth
                    || !string.Equals(restoredDigest, stored.PixelSha256, StringComparison.Ordinal))

                verified = true;

            Report(progress, 72, Locale.T("保存无损 PNG"));
            await EncodeAsync(transformed, temporary, width, height, bitDepth, token).ConfigureAwait(false);
            if (request.Operation == ImageObfuscationOperation.Obfuscate)
                ImageObfuscationPng.Add(temporary, new(1, ModeName(mode), iterations, width, height, bitDepth, originalDigest), token);
            Report(progress, 85, Locale.T("校验输出像素"));
            var outputInfo = ReadPng(temporary, token);
            if (outputInfo.Width != width || outputInfo.Height != height || outputInfo.BitDepth != bitDepth
                || outputInfo.IsAnimated || (request.Operation == ImageObfuscationOperation.Obfuscate && outputInfo.Marker is null))
                throw Failure("output_validation", Locale.T("生成的 PNG 没有通过格式校验。"), Locale.T("请重试；原图片保持不变。"));
            var outputPixels = await DecodeAsync(temporary, width, height, bitDepth, token).ConfigureAwait(false);
            if (!outputPixels.AsSpan().SequenceEqual(transformed))
                throw Failure("output_pixels", Locale.T("生成的 PNG 像素没有通过无损校验。"), Locale.T("请重试；原图片保持不变。"));

            statistics = new(sourceBytes, new FileInfo(temporary).Length, width, height, mode, iterations, verified);
            token.ThrowIfCancellationRequested();

                + (request.Operation == ImageObfuscationOperation.Obfuscate ? "_混淆" : "_解混淆"), token);
            temporary = null;
            log?.Write($"Image obfuscation completed: task={request.Id:N}, operation={request.Operation}, mode={mode}, iterations={iterations}, verified={verified}");
            Report(progress, 100, Locale.T("完成"));
            return new(JobStatus.Completed, completed, null, statistics);
        }
        catch (OperationCanceledException)
        {
            if (completed is not null) return new(JobStatus.Completed, completed, null, statistics);
            return cancellationToken.IsCancellationRequested ? new(JobStatus.Cancelled, null, null)
                : new(JobStatus.Failed, null, Failure("timeout", Locale.T("图片处理超过了时间限制。"), Locale.T("请减少处理次数或换一张较小的图片。")).Error);
        }
        catch (ConversionEngineException error)
        { return completed is null ? new(JobStatus.Failed, null, error.Error) : new(JobStatus.Completed, completed, null, statistics); }
        catch (Exception error)
        {
            if (completed is not null) return new(JobStatus.Completed, completed, null, statistics);
            var failure = error switch
            {
                UnauthorizedAccessException => Failure("access", Locale.T("没有权限读取图片或保存结果。"), Locale.T("请选择有读写权限的文件夹。"), error),
                InvalidDataException => Failure("invalid", Locale.T("图片数据或混淆标记不完整。"), Locale.T("请使用完整且未经修改的图片。"), error),
                IOException => Failure("io", Locale.T("图片读取或保存时遇到了问题。"), Locale.T("请检查文件占用、磁盘空间和保存位置。"), error),
                _ => Failure("unexpected", Locale.T("这张图片暂时无法处理。"), Locale.T("请确认图片完整后重试；可以查看高级详情。"), error)
            };
            log?.Write("Image obfuscation failed.", error.ToString());
            return new(JobStatus.Failed, null, failure.Error);
        }
        finally { DisposeSnapshot(snapshot, snapshotPath); Cleanup(normalizedPath); Cleanup(temporary); }
    }

using System.Buffers.Binary;
using System.Text;
using System.Text.Json;
using CatFileAssistant.Core;

namespace CatFileAssistant.Infrastructure;

internal sealed record ImageObfuscationMarker(int Version, string Mode, int Iterations,
    int Width, int Height, int BitDepth, string PixelSha256)
{
    public ImageObfuscationMode ParsedMode => Mode switch
    {
        "gilbert" => ImageObfuscationMode.Gilbert, "block" => ImageObfuscationMode.Block,
        "pixel" => ImageObfuscationMode.Pixel, _ => throw new InvalidDataException("Unknown image obfuscation mode.")
    };
}

internal sealed record ObfuscationPngInfo(int Width, int Height, int BitDepth, bool HasAlpha,
    bool IsAnimated, ImageObfuscationMarker? Marker);

/// <summary>Only bounded, non-secret application metadata is stored. Pixel decoding belongs to FFmpeg.</summary>
internal static class ImageObfuscationPng
{
    private static readonly byte[] Signature = [137, 80, 78, 71, 13, 10, 26, 10];
    private static readonly byte[] Keyword = Encoding.ASCII.GetBytes("CatFileAssistant.Obfuscation\0");
    private static readonly JsonSerializerOptions JsonOptions = new() { PropertyNamingPolicy = JsonNamingPolicy.CamelCase };
    private const int MaximumMarkerBytes = 4096;

    public static ObfuscationPngInfo Read(Stream stream, CancellationToken token)
    {
        stream.Position = 0;
        Span<byte> signature = stackalloc byte[8]; stream.ReadExactly(signature);
        if (!signature.SequenceEqual(Signature)) throw new InvalidDataException("Invalid PNG signature.");
        var width = 0; var height = 0; var bitDepth = 0; var alpha = false;
        var headerSeen = false; var imageSeen = false; var animated = false; var ended = false;
        ImageObfuscationMarker? marker = null;
        Span<byte> chunk = stackalloc byte[8]; Span<byte> crc = stackalloc byte[4];
        var chunks = 0;
        while (stream.Position + 12 <= stream.Length)
        {
            token.ThrowIfCancellationRequested();
            if (++chunks > 100_000) throw new InvalidDataException("Too many PNG chunks.");
            stream.ReadExactly(chunk);
            var length = BinaryPrimitives.ReadUInt32BigEndian(chunk[..4]);
            var payload = stream.Position;
            var end = checked(payload + length);
            if (end + 4 > stream.Length) throw new InvalidDataException("Truncated PNG chunk.");
            var kind = chunk[4..];
            if (!headerSeen && !kind.SequenceEqual("IHDR"u8)) throw new InvalidDataException("PNG IHDR must be first.");
            if (kind.SequenceEqual("IHDR"u8))
            {
                if (headerSeen || length != 13) throw new InvalidDataException("Invalid PNG IHDR.");
                var data = new byte[13]; stream.ReadExactly(data); stream.ReadExactly(crc);
                ValidateCrc(kind, data, crc);
                width = checked((int)BinaryPrimitives.ReadUInt32BigEndian(data.AsSpan(0, 4)));
                height = checked((int)BinaryPrimitives.ReadUInt32BigEndian(data.AsSpan(4, 4)));
                bitDepth = data[8];
                alpha = data[9] is 4 or 6;
                if (width <= 0 || height <= 0 || bitDepth is not (1 or 2 or 4 or 8 or 16))
                    throw new InvalidDataException("Invalid PNG dimensions or bit depth.");
                headerSeen = true;
            }
            else if (kind.SequenceEqual("tEXt"u8))
            {
                var prefix = new byte[(int)Math.Min(length, (uint)Keyword.Length)];
                stream.ReadExactly(prefix);
                if (prefix.AsSpan().SequenceEqual(Keyword))
                {
                    if (marker is not null || length > MaximumMarkerBytes)
                        throw new InvalidDataException("Duplicate or oversized image obfuscation metadata.");
                    var data = new byte[(int)length]; prefix.CopyTo(data, 0);
                    stream.ReadExactly(data.AsSpan(prefix.Length)); stream.ReadExactly(crc);
                    ValidateCrc(kind, data, crc);
                    try
                    {
                        marker = JsonSerializer.Deserialize<ImageObfuscationMarker>(data.AsSpan(Keyword.Length), JsonOptions)
                            ?? throw new InvalidDataException("Empty image obfuscation metadata.");
                    }
                    catch (JsonException error) { throw new InvalidDataException("Invalid image obfuscation metadata.", error); }
                    ValidateMarker(marker, width, height, bitDepth);
                }
            }
            else if (kind.SequenceEqual("acTL"u8)) animated = true;
            else if (kind.SequenceEqual("tRNS"u8)) alpha = true;
            else if (kind.SequenceEqual("IDAT"u8)) imageSeen = true;
            else if (kind.SequenceEqual("IEND"u8))
            {
                if (length != 0) throw new InvalidDataException("Invalid PNG IEND.");
                ended = true;
            }
            stream.Position = end + 4;
            if (ended) break;
        }
        if (!headerSeen || !imageSeen || !ended) throw new InvalidDataException("Incomplete PNG image.");
        return new(width, height, bitDepth > 8 ? 16 : 8, alpha, animated, marker);
    }

    private static void ValidateMarker(ImageObfuscationMarker marker, int width, int height, int pngDepth)
    {
        if (marker.Version != 1 || marker.Iterations is < 1 or > 10
            || marker.Width != width || marker.Height != height || marker.BitDepth is not (8 or 16)
            || marker.BitDepth != pngDepth || marker.PixelSha256 is null || marker.PixelSha256.Length != 64
            || marker.PixelSha256.Any(value => value is not (>= '0' and <= '9') and not (>= 'a' and <= 'f')))
            throw new InvalidDataException("Unsupported or inconsistent image obfuscation metadata.");
        _ = marker.ParsedMode;
    }

    public static void Add(string path, ImageObfuscationMarker marker, CancellationToken token)
    {
        token.ThrowIfCancellationRequested();
        var json = JsonSerializer.SerializeToUtf8Bytes(marker, JsonOptions);
        var payload = new byte[Keyword.Length + json.Length];
        Keyword.CopyTo(payload, 0); json.CopyTo(payload, Keyword.Length);
        if (payload.Length > MaximumMarkerBytes) throw new InvalidDataException("Image metadata exceeds its limit.");
        using var file = new FileStream(path, FileMode.Open, FileAccess.ReadWrite, FileShare.None);
        Span<byte> trailer = stackalloc byte[12];
        if (file.Length < 20) throw new InvalidDataException("PNG output is incomplete.");
        file.Position = file.Length - 12; file.ReadExactly(trailer);
        if (!trailer[..8].SequenceEqual(new byte[] { 0, 0, 0, 0, (byte)'I', (byte)'E', (byte)'N', (byte)'D' }))
            throw new InvalidDataException("PNG output has no final IEND.");
        file.Position -= 12;
        Span<byte> header = stackalloc byte[8];
        BinaryPrimitives.WriteUInt32BigEndian(header[..4], (uint)payload.Length); "tEXt"u8.CopyTo(header[4..]);
        file.Write(header); file.Write(payload);
        Span<byte> crc = stackalloc byte[4];
        BinaryPrimitives.WriteUInt32BigEndian(crc, Crc32(header[4..], payload)); file.Write(crc);
        file.Write(trailer); file.Flush(flushToDisk: true);
    }

    private static void ValidateCrc(ReadOnlySpan<byte> kind, ReadOnlySpan<byte> data, ReadOnlySpan<byte> encoded)
    {
        if (Crc32(kind, data) != BinaryPrimitives.ReadUInt32BigEndian(encoded))
            throw new InvalidDataException("PNG metadata checksum is invalid.");
    }

    private static uint Crc32(ReadOnlySpan<byte> kind, ReadOnlySpan<byte> data)
    {
        var crc = uint.MaxValue;
        foreach (var value in kind) crc = Update(crc, value);
        foreach (var value in data) crc = Update(crc, value);
        return ~crc;

        static uint Update(uint crc, byte value)
        {
            crc ^= value;
            for (var bit = 0; bit < 8; bit++) crc = (crc >> 1) ^ ((crc & 1) != 0 ? 0xedb88320u : 0);
            return crc;
        }
    }
}

def gilbert2d(width, height):
    """
    Generalized Hilbert ('gilbert') space-filling curve for arbitrary-sized
    2D rectangular grids. Generates discrete 2D coordinates to fill a rectangle
    of size (width x height).
    """

    if width >= height:
        yield from generate2d(0, 0, width, 0, 0, height)
    else:
        yield from generate2d(0, 0, 0, height, width, 0)


def sgn(x):
    return -1 if x < 0 else (1 if x > 0 else 0)


def generate2d(x, y, ax, ay, bx, by):

    w = abs(ax + ay)
    h = abs(bx + by)

    (dax, day) = (sgn(ax), sgn(ay)) # unit major direction
    (dbx, dby) = (sgn(bx), sgn(by)) # unit orthogonal direction

    if h == 1:
        # trivial row fill
        for i in range(0, w):
            yield(x, y)
            (x, y) = (x + dax, y + day)
        return

    if w == 1:
        # trivial column fill
        for i in range(0, h):
            yield(x, y)
            (x, y) = (x + dbx, y + dby)
        return

    (ax2, ay2) = (ax//2, ay//2)
    (bx2, by2) = (bx//2, by//2)

    w2 = abs(ax2 + ay2)
    h2 = abs(bx2 + by2)

    if 2*w > 3*h:
        if (w2 % 2) and (w > 2):
            # prefer even steps
            (ax2, ay2) = (ax2 + dax, ay2 + day)

        # long case: split in two parts only
        yield from generate2d(x, y, ax2, ay2, bx, by)
        yield from generate2d(x+ax2, y+ay2, ax-ax2, ay-ay2, bx, by)

    else:
        if (h2 % 2) and (h > 2):
            # prefer even steps
            (bx2, by2) = (bx2 + dbx, by2 + dby)

        # standard case: one step up, one long horizontal, one step down
        yield from generate2d(x, y, bx2, by2, ax2, ay2)
        yield from generate2d(x+bx2, y+by2, ax, ay, bx-bx2, by-by2)
        yield from generate2d(x+(ax-dax)+(bx2-dbx), y+(ay-day)+(by2-dby),
                              -bx2, -by2, -(ax-ax2), -(ay-ay2))

using System.Buffers.Binary;

/// <summary>Version 1 public, password-free permutations. No security or third-party format claim.</summary>
internal static class ImageObfuscationTransforms
{
    private const int BlockSize = 32;
    private readonly record struct Tile(int X, int Y, int Width, int Height);

    public static byte[] Apply(byte[] pixels, int width, int height, int bytesPerPixel,
        ImageObfuscationMode mode, int iterations, bool inverse,
        Action<double> report, CancellationToken token)
    {
        var count = checked(width * height);
        var destination = new byte[pixels.Length];
        var path = mode == ImageObfuscationMode.Gilbert ? GilbertCurve.CreatePath(width, height, token) : null;
        var permutation = mode == ImageObfuscationMode.Pixel ? Permutation(count, "pixel", width, height, token) : null;
        List<(List<Tile> Tiles, int[] Permutation)>? tileGroups = null;
        if (mode == ImageObfuscationMode.Block)
        {
            var groups = new Dictionary<(int Width, int Height), List<Tile>>();
            for (var y = 0; y < height; y += BlockSize)
                for (var x = 0; x < width; x += BlockSize)
                {
                    var size = (Width: Math.Min(BlockSize, width - x), Height: Math.Min(BlockSize, height - y));
                    if (!groups.TryGetValue(size, out var tiles)) groups[size] = tiles = [];
                    tiles.Add(new(x, y, size.Width, size.Height));
                }
            tileGroups = groups.Select(group => (group.Value, Permutation(group.Value.Count,
                "block:" + group.Key.Width.ToString(CultureInfo.InvariantCulture) + "x" + group.Key.Height.ToString(CultureInfo.InvariantCulture),
                width, height, token))).ToList();
        }
        var shift = (int)Math.Floor(count * 0.6180339887498949d);
        for (var iteration = 0; iteration < iterations; iteration++)
        {
            token.ThrowIfCancellationRequested();
            if (mode == ImageObfuscationMode.Block)
            {
                var copied = 0;
                foreach (var group in tileGroups!)
                    for (var index = 0; index < group.Tiles.Count; index++)
                    {
                        var to = group.Tiles[inverse ? group.Permutation[index] : index];
                        var from = group.Tiles[inverse ? index : group.Permutation[index]];
                        for (var row = 0; row < to.Height; row++)
                        {
                            token.ThrowIfCancellationRequested();
                            pixels.AsSpan(((from.Y + row) * width + from.X) * bytesPerPixel, from.Width * bytesPerPixel)
                                .CopyTo(destination.AsSpan(((to.Y + row) * width + to.X) * bytesPerPixel));
                        }
                        copied += to.Width * to.Height;
                        if ((index & 127) == 0) report((iteration + (double)copied / count) / iterations);
                    }
            }
            else
            {
                for (var index = 0; index < count; index++)
                {
                    if ((index & 4095) == 0)
                    {
                        token.ThrowIfCancellationRequested();
                        report((iteration + (double)index / count) / iterations);
                    }
                    int from, to;
                    if (mode == ImageObfuscationMode.Gilbert)
                    {
                        to = path![index];
                        from = path[(index + (inverse ? count - shift : shift)) % count];
                    }
                    else
                    {
                        to = inverse ? permutation![index] : index;
                        from = inverse ? index : permutation![index];
                    }
                    pixels.AsSpan(from * bytesPerPixel, bytesPerPixel)
                        .CopyTo(destination.AsSpan(to * bytesPerPixel, bytesPerPixel));
                }
            }
            (pixels, destination) = (destination, pixels);
            report((double)(iteration + 1) / iterations);
        }
        return pixels;
    }

    private static int[] Permutation(int count, string domain, int width, int height, CancellationToken token)
    {
        var permutation = new int[count];
        for (var index = 0; index < count; index++)
        {
            if ((index & 4095) == 0) token.ThrowIfCancellationRequested();
            permutation[index] = index;
        }
        var random = new PublicRandomStream(domain, width, height);
        for (var index = count - 1; index > 0; index--)
        {
            if ((index & 4095) == 0) token.ThrowIfCancellationRequested();
            var chosen = (int)random.Next((uint)(index + 1));
            (permutation[index], permutation[chosen]) = (permutation[chosen], permutation[index]);
        }
        return permutation;
    }

    private sealed class PublicRandomStream
    {
        private readonly byte[] input;
        private readonly byte[] block = new byte[32];
        private ulong counter;
        private int position = 32;

        public PublicRandomStream(string domain, int width, int height)
        {
            var seed = Encoding.UTF8.GetBytes("CatFileAssistant.ImageObfuscation.v1|" + domain + "|"
                + width.ToString(CultureInfo.InvariantCulture) + "|" + height.ToString(CultureInfo.InvariantCulture));
            input = new byte[seed.Length + 8]; seed.CopyTo(input, 0);
        }

        public uint Next(uint bound)
        {
            var threshold = unchecked(0u - bound) % bound;
            uint value;
            do { value = NextUInt32(); } while (value < threshold);
            return value % bound;
        }

        private uint NextUInt32()
        {
            if (position == block.Length)
            {
                BinaryPrimitives.WriteUInt64LittleEndian(input.AsSpan(input.Length - 8), counter++);
                SHA256.HashData(input, block);
                position = 0;
            }
            var value = BinaryPrimitives.ReadUInt32LittleEndian(block.AsSpan(position, 4)); position += 4;
            return value;
        }
    }

using System.Runtime.InteropServices;
using System.Text.Json;
using Microsoft.Win32.SafeHandles;
using CatFileAssistant.Ownership;

namespace CatFileAssistant.Infrastructure;

/// <summary>Exact, identity-checked receipts for transient files. Never scans output folders.</summary>
public static class OwnedArtifacts
{
    private sealed record Receipt(string Path, string Identity, bool Directory);
    private static readonly object gate = new();
    private static string root = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "CatFileAssistant");
    private const string LedgerName = "owned-temporary-files.json";
    public static void Configure(string dataRoot) { lock (gate) root = Path.GetFullPath(dataRoot); }

    public static string CreateArchiveJournal(string workRoot)
    {
        lock (gate)
        {
            if (!SafePath(root) || !SafePath(workRoot)) throw new IOException("Archive artifact data path is unsafe.");
            System.IO.Directory.CreateDirectory(root);
            var journal = Path.Combine(root, ".cat-archive-receipts-" + Guid.NewGuid().ToString("N") + ".jsonl");
            ArchiveTemporaryJournal.Create(journal, workRoot);
            try { Register(journal); return journal; }
            catch { File.Delete(journal); throw; }
        }
    }

    public static void Register(string path, bool directory = false)
    {
        lock (gate)
        {
            path = Path.GetFullPath(path);
            if (!IsTransientName(path, root) || !SafePath(path)) throw new IOException("Temporary artifact has an unsafe path.");
            using var handle = OpenIdentity(path, directory, deleteAccess: false);
            var receipt = new Receipt(path, Identity(handle), directory);
            var entries = Read(root).Where(item => File.Exists(item.Path) || System.IO.Directory.Exists(item.Path)).ToList();
            entries.RemoveAll(item => string.Equals(item.Path, path, StringComparison.OrdinalIgnoreCase));
            entries.Add(receipt);
            Save(root, entries);
        }
    }

    public static IReadOnlyList<string> Cleanup(string dataRoot)
    {
        lock (gate)
        {
            dataRoot = Path.GetFullPath(dataRoot);
            if (!SafePath(dataRoot)) return new[] { dataRoot };
            var retained = new List<Receipt>();
            foreach (var item in Read(dataRoot).OrderByDescending(item => item.Path.Length))
            {
                if (!File.Exists(item.Path) && !System.IO.Directory.Exists(item.Path)) continue;
                try
                {
                    if (!Path.IsPathFullyQualified(item.Path) || !IsTransientName(item.Path, dataRoot) || !SafePath(item.Path)) { retained.Add(item); continue; }
                    using var handle = OpenIdentity(item.Path, item.Directory, deleteAccess: true);
                    if (Identity(handle) != item.Identity) { retained.Add(item); continue; }
                    if (!item.Directory && Path.GetFileName(item.Path).StartsWith(".cat-archive-receipts-", StringComparison.Ordinal) && !ArchiveTemporaryJournal.Cleanup(item.Path))
                    { retained.Add(item); continue; }
                    if (item.Directory && System.IO.Directory.EnumerateFileSystemEntries(item.Path).Any()) { retained.Add(item); continue; }
                    // Delete the verified handle, so replacing the pathname cannot target another file.
                    var disposition = new FileDisposition { Delete = true };
                    if (!SetFileInformationByHandle(handle, 4, ref disposition, (uint)Marshal.SizeOf<FileDisposition>()))
                        throw new IOException("Owned artifact could not be removed.");
                }
                catch (Exception error) when (error is IOException or UnauthorizedAccessException or System.ComponentModel.Win32Exception)
                { retained.Add(item); }
            }
            Save(dataRoot, retained);
            return retained.Select(item => item.Path).ToArray();
        }
    }

    private static bool IsTransientName(string path, string dataRoot)
    {
        var name = Path.GetFileName(path);
        if (name.StartsWith(".cat-", StringComparison.Ordinal)) return true;
        return string.Equals(Path.GetDirectoryName(path), dataRoot, StringComparison.OrdinalIgnoreCase) &&
            name.EndsWith(".tmp", StringComparison.OrdinalIgnoreCase) && (name.StartsWith("settings-", StringComparison.Ordinal) || name.StartsWith("pet-state-", StringComparison.Ordinal));
    }
    public static bool SafePath(string path)
    {
        try
        {
            for (var current = Path.GetFullPath(path); current is not null; current = Path.GetDirectoryName(current))
                if ((File.Exists(current) || System.IO.Directory.Exists(current)) && (File.GetAttributes(current) & FileAttributes.ReparsePoint) != 0) return false;
            return true;
        }
        catch (Exception error) when (error is IOException or UnauthorizedAccessException or ArgumentException) { return false; }
    }
    private static List<Receipt> Read(string dataRoot)
    {
        var path = Path.Combine(dataRoot, LedgerName);
        if (!File.Exists(path)) return new();
        if (!SafePath(path)) throw new IOException("Artifact receipt is a reparse point.");
        var entries = JsonSerializer.Deserialize<List<Receipt>>(File.ReadAllText(path)) ?? throw new InvalidDataException("Invalid artifact receipt.");
        if (entries.Any(item => item is null || string.IsNullOrWhiteSpace(item.Path) || string.IsNullOrWhiteSpace(item.Identity) || !Path.IsPathFullyQualified(item.Path)))
            throw new InvalidDataException("Invalid artifact receipt entry.");
        return entries;
    }
    private static void Save(string dataRoot, List<Receipt> entries)
    {
        if (!SafePath(dataRoot)) throw new IOException("Artifact data path is unsafe.");
        var path = Path.Combine(dataRoot, LedgerName);
        if (entries.Count == 0) { if (File.Exists(path) && SafePath(path)) File.Delete(path); return; }
        System.IO.Directory.CreateDirectory(dataRoot);
        var temporary = Path.Combine(dataRoot, LedgerName + ".tmp");
        if (File.Exists(temporary) && !SafePath(temporary)) throw new IOException("Artifact receipt temporary path is unsafe.");
        try
        {
            File.WriteAllText(temporary, JsonSerializer.Serialize(entries));
            File.Move(temporary, path, true);
        }
        finally { if (File.Exists(temporary) && SafePath(temporary)) File.Delete(temporary); }
    }
    private static SafeFileHandle OpenIdentity(string path, bool directory, bool deleteAccess)
    {
        var handle = CreateFile(path, 0x80u | (deleteAccess ? 0x10000u : 0u), 7, IntPtr.Zero, 3, 0x00200000u | (directory ? 0x02000000u : 0u), IntPtr.Zero);
        if (handle.IsInvalid) { handle.Dispose(); throw new IOException("Cannot open artifact identity.", Marshal.GetExceptionForHR(Marshal.GetHRForLastWin32Error())); }
        return handle;
    }
    private static string Identity(SafeFileHandle handle)
    {
        if (!GetFileInformationByHandle(handle, out var info)) throw new IOException("Cannot read artifact identity.");
        return $"{info.Volume:X8}-{info.IndexHigh:X8}{info.IndexLow:X8}-{info.CreationHigh:X8}{info.CreationLow:X8}";
    }
    [StructLayout(LayoutKind.Sequential)] private struct FileInfo
    { public uint Attributes, CreationLow, CreationHigh, AccessLow, AccessHigh, WriteLow, WriteHigh, Volume, SizeHigh, SizeLow, Links, IndexHigh, IndexLow; }
    [StructLayout(LayoutKind.Sequential)] private struct FileDisposition { [MarshalAs(UnmanagedType.Bool)] public bool Delete; }
    [DllImport("kernel32.dll", CharSet = CharSet.Unicode, SetLastError = true)] private static extern SafeFileHandle CreateFile(string name, uint access, uint share, IntPtr security, uint creation, uint flags, IntPtr template);

    [DllImport("kernel32.dll", SetLastError = true)] [return: MarshalAs(UnmanagedType.Bool)] private static extern bool SetFileInformationByHandle(SafeFileHandle handle, int type, ref FileDisposition information, uint size);
}

            _ => Fail("recovery.worker", Locale.T("密码恢复引擎未能完成任务。"), Locale.T("请检查压缩包完整性和可用内存后重试。"))

    }

    private static bool IsRecoveryArchive(string source) => Regex.IsMatch(Path.GetFileName(source),
        @"\.(?:zip|7z|rar)(?:\.\d{3,})?$", RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);

    private static IEnumerable<string> RelatedSources(string source)
    {
        yield return source;
        var name = Path.GetFileName(source); var directory = Path.GetDirectoryName(source)!;
        var numbered = Regex.Match(name, @"^(.*\.(?:7z|zip))\.\d{3,}$", RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);
        var part = Regex.Match(name, @"^(.*)\.part\d+\.rar$", RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);
        var pattern = numbered.Success ? "^" + Regex.Escape(numbered.Groups[1].Value) + @"\.\d{3,}$"
            : part.Success ? "^" + Regex.Escape(part.Groups[1].Value) + @"\.part\d+\.rar$"
            : name.EndsWith(".rar", StringComparison.OrdinalIgnoreCase) ? "^" + Regex.Escape(Path.GetFileNameWithoutExtension(name)) + @"\.[r-z]\d{2}$"
            : name.EndsWith(".zip", StringComparison.OrdinalIgnoreCase) ? "^" + Regex.Escape(Path.GetFileNameWithoutExtension(name)) + @"\.z\d+$" : null;
        if (pattern is null) yield break;
        foreach (var sibling in Directory.EnumerateFiles(directory))
            if (!string.Equals(sibling, source, StringComparison.OrdinalIgnoreCase)
                && Regex.IsMatch(Path.GetFileName(sibling), pattern, RegexOptions.IgnoreCase | RegexOptions.CultureInvariant)) yield return sibling;
    }
    private static void RejectReparseAncestors(string path)
    {
        for (var current = Path.GetFullPath(path); !string.IsNullOrEmpty(current); current = Path.GetDirectoryName(current))
            if ((File.GetAttributes(current) & FileAttributes.ReparsePoint) != 0) throw new IOException("Archive links are not supported.");
    }

    public static decimal EarnedFor(long ticks)
    {
        var seconds = Math.Max(0, ticks / TimeSpan.TicksPerSecond);
        return decimal.Floor(seconds * (decimal)WorkCoinsPerMinute * 100 / 60) / 100;

    private bool HasPendingRewards => pendingBatches.Count > 0 || pendingStop || checkpointPending || migrationPending;
    private bool CanWork => !loadFailed && state.WorkRequested && !systemPaused && !pendingStop && !checkpointPending;
    private PetTransactionResult? TransactionsBlocked => loadFailed

        : HasPendingRewards ? new(false, "奖励还在等待保存，保存成功后再进行背包交易。") : null;

    private PetTransactionResult Execute(Func<PetTransactionResult> action)
    {
        PetTransactionResult result;
        lock (gate)
        {
            CaptureWorkElapsed();
            result = action();
        }
        // A view's failed event handler must never turn a committed transaction into a failure.
        if (StateChanged is { } handlers)
            foreach (EventHandler handler in handlers.GetInvocationList())
                try { handler(this, EventArgs.Empty); } catch (Exception) { }
        return result;
    }

    private void CaptureWorkElapsed()
    {
        var now = clock.GetTimestamp();
        var elapsed = clock.GetElapsedTime(lastTimestamp, now);
        lastTimestamp = now;
        if (!CanWork || elapsed.Ticks <= 0) return;
        elapsedTicks = elapsed.Ticks > TimeSpan.MaxValue.Ticks - elapsedTicks ? TimeSpan.MaxValue.Ticks : elapsedTicks + elapsed.Ticks;
        elapsedSinceCheckpoint += elapsed.TotalMilliseconds;
    }

    private PetTransactionResult RetryRewardsCore()
    {
        if (loadFailed) return new(false, lastError ?? "桌宠存档暂时无法读取，奖励等待保存。");
        decimal earned = 0;
        foreach (var batch in pendingBatches.ToArray())
        {
            if (state.SettledBatchIds.Contains(batch.Key)) { pendingBatches.Remove(batch.Key); continue; }
            var next = Candidate();
            // Queue row IDs can be reused by a genuine later operation. The batch receipt
            // deduplicates callbacks; the set deduplicates repeated passes within this batch.
            if (batch.Value.Count > int.MaxValue / FileRewardCoins)
                return new(false, "这批任务的奖励数量超过上限，奖励等待保存。", earned);
            var reward = batch.Value.Count * FileRewardCoins;
            if (next.Coins > int.MaxValue - reward) return new(false, "喵喵币余额已达到上限，奖励等待保存。");
            next.Coins += reward;
            next.SettledBatchIds.Add(batch.Key);
            TrimReceipts(next);
            if (state.LastSettlementSequence == long.MaxValue)
                return new(false, "桌宠结算序号已达到上限，奖励等待保存。", earned);
            next.LastSettlementSequence = state.LastSettlementSequence + 1;
            if (!TryCommit(next, out var error)) return new(false, error, earned);
            pendingBatches.Remove(batch.Key);
            earned += reward;
        }
        if (pendingStop)
        {
            var next = Candidate();
            var delta = next.Coins - state.Coins;
            next.WorkRequested = false;
            next.WorkPaused = false;
            next.LastWorkSession = new(next.WorkSessionId, TimeSpan.FromTicks(elapsedTicks), EarnedFor(elapsedTicks));
            if (!TryCommit(next, out var error)) return new(false, error, earned);
            pendingStop = false;
            migrationPending = false;
            earned += delta;
            return new(true, Locale.T("本次工作已结算。"), earned, next.LastWorkSession);
        }
        if (checkpointPending || migrationPending)
        {
            var next = Candidate();
            var delta = next.Coins - state.Coins;
            if (!TryCommit(next, out var error)) return new(false, error, earned);
            migrationPending = false;
            earned += delta;
        }
        return new(true, earned > 0 ? Locale.F($"收到了 {earned:0.##} 喵喵币，谢谢你和猫咪一起努力。") : Locale.T("奖励已保存。"), earned);
    }

    private PetTransactionResult FlushCore()
    {
        if (HasPendingRewards) return RetryRewardsCore();
        if (loadFailed) return new(false, lastError ?? "桌宠存档暂时无法读取。");
        if (state.WorkElapsedTicks == elapsedTicks) return new(true, "");
        var next = Candidate();
        return CommitResult(next, Locale.T("工作进度已保存。"), next.Coins - state.Coins);
    }

    private PetStateDocument Candidate()
    {
        var earned = EarnedFor(elapsedTicks);
        return new()
        {
            Version = 2, Coins = state.Coins + Math.Max(0, earned - state.WorkCreditedCoins),
            Inventory = new(state.Inventory), EquippedAccessories = new(state.EquippedAccessories),
            WorkRequested = state.WorkRequested, WorkPaused = false, WorkSessionId = state.WorkSessionId,
            WorkElapsedTicks = elapsedTicks, WorkCreditedCoins = earned, LastWorkSession = state.LastWorkSession,
            Sequence = state.Sequence, LastSettlementSequence = state.LastSettlementSequence,
            SettledBatchIds = [.. state.SettledBatchIds]
        };
    }

    private PetTransactionResult CommitResult(PetStateDocument next, string message, decimal coinsChanged = 0)
        => TryCommit(next, out var error) ? new(true, message, coinsChanged) : new(false, error);

    private bool TryCommit(PetStateDocument next, out string error)
    {
        try
        {
            if (next.Coins < 0 || next.Coins > int.MaxValue) throw new OverflowException("Wallet limit.");
            next.Sequence = checked(state.Sequence + 1);
            persistence.WriteAtomically(FilePath, JsonSerializer.Serialize(next, json));
            state = next;
            elapsedSinceCheckpoint = 0;
            checkpointPending = false;
            lastError = null;
            error = "";
            return true;
        }
        catch (Exception exception) when (exception is IOException or UnauthorizedAccessException or JsonException or OverflowException)
        {
            checkpointPending = state.WorkRequested || pendingStop || migrationPending;
            error = lastError = Locale.T("桌宠存档暂时无法保存，这次交易没有确认；待保存的奖励会自动重试。");
            return false;
        }
    }

    private void Load()
    {
        try
        {
            var contents = persistence.Read(FilePath) ?? persistence.Read(FilePath + ".bak");
            if (contents is null) return;
            state = Deserialize(contents);
        }
        catch (Exception exception) when (exception is IOException or UnauthorizedAccessException or JsonException or InvalidDataException)
        {
            // A damaged wallet must never silently become a fresh funded account.
            // Atomic file replacement keeps a previous confirmed revision for recovery.
            try
            {
                var backup = persistence.Read(FilePath + ".bak");
                if (backup is null) throw new InvalidDataException("Missing wallet backup.");
                state = Deserialize(backup);
                lastError = "已读取上一次保存的桌宠存档，请保留当前存档以便排查。";
            }
            catch (Exception backupException) when (backupException is IOException or UnauthorizedAccessException or JsonException or InvalidDataException)
            {
                loadFailed = true;
                state = new PetStateDocument { Coins = 0 };
                lastError = "桌宠存档暂时无法读取，账户交易已暂停；没有覆盖原存档。";
            }
        }
    }

    private PetStateDocument Deserialize(string contents)
    {
        // Missing wallet fields must not receive funded account defaults. In particular,
        // an empty object is a damaged existing wallet, not a newly created account.
        using var document = JsonDocument.Parse(contents);
        if (document.RootElement.ValueKind != JsonValueKind.Object
            || new[] { "Version", "Coins", "Inventory", "EquippedAccessories", "WorkRequested",
                "Sequence", "LastSettlementSequence", "SettledBatchIds" }
                .Any(name => !document.RootElement.TryGetProperty(name, out _)))
            throw new InvalidDataException("Incomplete wallet.");
        var loaded = JsonSerializer.Deserialize<PetStateDocument>(contents, json) ?? throw new InvalidDataException("Empty wallet.");
        if (loaded.Version is not (1 or 2) || loaded.Coins < 0 || loaded.Coins > int.MaxValue || decimal.Round(loaded.Coins, 2) != loaded.Coins
            || loaded.Sequence < 0 || loaded.LastSettlementSequence < 0 || loaded.LastSettlementSequence > loaded.Sequence
            || loaded.Inventory is null || loaded.EquippedAccessories is null || loaded.SettledBatchIds is null)
            throw new InvalidDataException("Invalid wallet.");
        if (loaded.Version == 1)
        {
            if (!document.RootElement.TryGetProperty("WorkRemainingMilliseconds", out _) || !document.RootElement.TryGetProperty("WorkPaused", out _)
                || !double.IsFinite(loaded.WorkRemainingMilliseconds) || loaded.WorkRemainingMilliseconds < 0
                || loaded.WorkRemainingMilliseconds > WorkDuration.TotalMilliseconds)
                throw new InvalidDataException("Invalid legacy work.");
            loaded.WorkElapsedTicks = loaded.WorkRequested
                ? TimeSpan.FromMilliseconds(WorkDuration.TotalMilliseconds - loaded.WorkRemainingMilliseconds).Ticks : 0;
            loaded.WorkCreditedCoins = 0;
            loaded.WorkSessionId = loaded.WorkRequested ? Guid.NewGuid() : Guid.Empty;
        }
        else
        {
            if (new[] { "WorkElapsedTicks", "WorkCreditedCoins", "WorkSessionId", "LastWorkSession" }
                .Any(name => !document.RootElement.TryGetProperty(name, out _))
                || loaded.WorkElapsedTicks < 0 || loaded.WorkElapsedTicks > TimeSpan.MaxValue.Ticks
                || loaded.WorkCreditedCoins < 0 || loaded.WorkCreditedCoins != EarnedFor(loaded.WorkElapsedTicks)
                || loaded.WorkRequested && loaded.WorkSessionId == Guid.Empty
                || loaded.LastWorkSession is { } summary && (summary.Duration < TimeSpan.Zero || summary.CoinsEarned != EarnedFor(summary.Duration.Ticks)))
                throw new InvalidDataException("Invalid cumulative work.");
        }
        foreach (var entry in loaded.Inventory)
        {
            var item = PetShopCatalog.Find(entry.Key);
            if (item is null || entry.Value <= 0 || (!item.Consumable && entry.Value != 1)) throw new InvalidDataException("Invalid inventory.");

using System.Diagnostics;
using CatFileAssistant.Core;
using System.Text;

namespace CatFileAssistant.Infrastructure;

internal sealed record ProcessOutcome(int ExitCode, string StandardError);
internal sealed record BinaryProcessOutcome(int ExitCode, string StandardError, byte[] StandardOutput);

internal static class ProcessRunner
{
    private const int MaximumErrorCharacters = 32 * 1024;

    /// <summary>Bounded binary pipes. The caller owns all bytes; no raw image is written to disk.</summary>
    public static async Task<BinaryProcessOutcome> RunBinaryAsync(string executable, IEnumerable<string> arguments,
        byte[]? standardInput, int maximumOutputBytes, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        if (maximumOutputBytes < 0) throw new ArgumentOutOfRangeException(nameof(maximumOutputBytes));
        var start = new ProcessStartInfo(executable)
        {
            UseShellExecute = false, CreateNoWindow = true,
            RedirectStandardInput = standardInput is not null,
            RedirectStandardOutput = true, RedirectStandardError = true,
            StandardErrorEncoding = Encoding.UTF8
        };
        foreach (var argument in arguments) start.ArgumentList.Add(argument);
        using var process = new Process { StartInfo = start };
        if (!process.Start()) throw new IOException(Locale.T("无法启动图片处理引擎。"));
        var errors = new StringBuilder();
        var stdout = ReadBoundedAsync(process.StandardOutput.BaseStream, maximumOutputBytes, cancellationToken);
        var stderr = DrainAsync(process.StandardError, line =>
        {
            errors.AppendLine(line);
            if (errors.Length > MaximumErrorCharacters) errors.Remove(0, errors.Length - MaximumErrorCharacters);
        });
        var stdin = standardInput is null ? Task.CompletedTask : SendInputAsync();
        var pipes = Task.WhenAll(stdout, stderr, stdin);
        var pipeFailure = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        foreach (var pipe in new Task[] { stdout, stderr, stdin })
            _ = pipe.ContinueWith(task =>
            {
                if (task.IsFaulted) pipeFailure.TrySetException(task.Exception!.InnerExceptions);
                else if (task.IsCanceled) pipeFailure.TrySetCanceled(cancellationToken);
            }, CancellationToken.None, TaskContinuationOptions.ExecuteSynchronously, TaskScheduler.Default);
        try
        {
            var exited = process.WaitForExitAsync(cancellationToken);
            // Pipe limits and write failures must terminate a producer promptly, even if it is still running.
            await Task.WhenAny(exited, pipeFailure.Task).ConfigureAwait(false);
            if (pipeFailure.Task.IsCompleted) await pipeFailure.Task.ConfigureAwait(false);
            await exited.ConfigureAwait(false);
            await pipes.ConfigureAwait(false);
            cancellationToken.ThrowIfCancellationRequested();
            return new(process.ExitCode, errors.ToString(), await stdout.ConfigureAwait(false));
        }
        catch
        {
            try { if (!process.HasExited) process.Kill(entireProcessTree: true); }
            catch (InvalidOperationException) { }
            catch (System.ComponentModel.Win32Exception) { }
            await process.WaitForExitAsync(CancellationToken.None).ConfigureAwait(false);
            try { await pipes.ConfigureAwait(false); }
            catch (Exception) { }
            if (pipeFailure.Task.IsFaulted) _ = pipeFailure.Task.Exception;
            cancellationToken.ThrowIfCancellationRequested();
            throw;
        }

        async Task SendInputAsync()
        {
            try
            {
                await process.StandardInput.BaseStream.WriteAsync(standardInput!, cancellationToken).ConfigureAwait(false);
                await process.StandardInput.BaseStream.FlushAsync(cancellationToken).ConfigureAwait(false);
            }
            finally { process.StandardInput.Close(); }
        }
    }

    private static async Task<byte[]> ReadBoundedAsync(Stream stream, int maximumBytes, CancellationToken token)
    {
        var bytes = new byte[maximumBytes];
        var used = 0;
        while (used < bytes.Length)
        {
            var read = await stream.ReadAsync(bytes.AsMemory(used, Math.Min(81920, bytes.Length - used)), token).ConfigureAwait(false);
            if (read == 0) return used == bytes.Length ? bytes : bytes.AsSpan(0, used).ToArray();
            used += read;
        }
        var extra = new byte[1];
        if (await stream.ReadAsync(extra, token).ConfigureAwait(false) != 0)
            throw new InvalidDataException("Image decoder exceeded its exact pixel buffer limit.");
        return bytes;
    }

    public static async Task<ProcessOutcome> RunAsync(string executable, IEnumerable<string> arguments, Action<string>? outputLine, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        var startInfo = new ProcessStartInfo(executable)
        {
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            StandardOutputEncoding = Encoding.UTF8,
            StandardErrorEncoding = Encoding.UTF8
        };
        foreach (var argument in arguments) startInfo.ArgumentList.Add(argument);
        using var process = new Process { StartInfo = startInfo };
        if (!process.Start()) throw new IOException(Locale.T("无法启动转换引擎。"));
        var errors = new StringBuilder();
        var stdout = DrainAsync(process.StandardOutput, outputLine);
        var stderr = DrainAsync(process.StandardError, line =>
        {
            errors.AppendLine(line);
            if (errors.Length > MaximumErrorCharacters) errors.Remove(0, errors.Length - MaximumErrorCharacters);
        });
        try
        {
            await process.WaitForExitAsync(cancellationToken).ConfigureAwait(false);
            await Task.WhenAll(stdout, stderr).ConfigureAwait(false);
            cancellationToken.ThrowIfCancellationRequested();
            return new ProcessOutcome(process.ExitCode, errors.ToString());
        }
        catch
        {
            // A task owns its process until it has exited and both pipes have been drained.
            try { if (!process.HasExited) process.Kill(entireProcessTree: true); }
            catch (InvalidOperationException) { }
            catch (System.ComponentModel.Win32Exception) { }
            await process.WaitForExitAsync(CancellationToken.None).ConfigureAwait(false);
            try { await Task.WhenAll(stdout, stderr).ConfigureAwait(false); }
            catch (IOException) { }
            throw;
        }
    }

    private static async Task DrainAsync(StreamReader reader, Action<string>? lineAction)
    {
        while (await reader.ReadLineAsync().ConfigureAwait(false) is { } line)
            lineAction?.Invoke(line);
    }
}

internal sealed record RecoveryShard(int Digits, string[]? Candidates, int RangeCount, int Lane, int Lanes,
    int? CpuFallbackPercent)
{
    public int Count => ((Candidates?.Length ?? RangeCount) + Lanes - 1 - Lane) / Lanes;
    public long? CompletionOrdinal(string password)
    {
        if (password.Length != Digits || password.Any(c => c is < '0' or > '9')) return null;
        if (Candidates is not null)
        {
            var index = Array.IndexOf(Candidates, password);
            return index >= 0 && index % Lanes == Lane ? (index - Lane) / Lanes + 1 : null;
        }
        return int.TryParse(password, NumberStyles.None, CultureInfo.InvariantCulture, out var number)
            && number < RangeCount && number % Lanes == Lane ? (number - Lane) / Lanes + 1 : null;
    }
}
internal sealed record RecoveryWorkerRequest(string Action, Guid Id, string SourcePath,

/// <summary>One shared job bounds every lane. Secrets travel only over anonymous pipes.</summary>
internal static class RecoveryWorkerPool
{
    private static readonly JsonSerializerOptions Json = new(JsonSerializerDefaults.Web);
    private const int MaximumFrameCharacters = 256 * 1024;

    internal static async Task<RecoveryPoolResult> RunStageAsync(string workerPath, string dllPath,
        ArchiveWorkerJob job, IReadOnlyList<RecoveryWorkerRequest> requests,
        Action<long, string, bool> onProgress, CancellationToken cancellationToken)
    {
        using var stop = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        var gate = new object(); var completed = new long[requests.Count]; var possible = false;
        RecoveryWorkerTerminal? found = null; string? failure = null;
        long lastReport = 0;
        void Update(int lane, long count, string stage, bool maybe, bool force = false)
        {
            lock (gate)
            {
                if (count < completed[lane] || count > requests[lane].Recovery.Count) throw new InvalidDataException();
                completed[lane] = count; possible |= maybe;
                var now = Environment.TickCount64;
                if (force || stage == "verifying" || now - lastReport >= 200)
                {
                    lastReport = now;
                    try { onProgress(completed.Sum(), stage, possible); } catch (Exception) { }
                }
            }
        }
        void Terminal(int lane, RecoveryWorkerTerminal result)
        {
            var shard = requests[lane].Recovery;
            if (result.Status is not ("found" or "exhausted" or "failed" or "cancelled")
                || result.CompletedCandidates < 0 || result.CompletedCandidates > shard.Count
                || result.Status == "exhausted" && result.CompletedCandidates != shard.Count
                || result.Status == "found" && (result.CompletedCandidates < 1 || result.RecoveredPassword is null
                    || shard.CompletionOrdinal(result.RecoveredPassword) != result.CompletedCandidates))
                throw new InvalidDataException();
            Update(lane, result.CompletedCandidates, "testing", result.PossiblePassword, true);
            lock (gate)
            {
                if (result.Status == "found") found ??= result;
                else if (result.Status == "failed") failure ??= result.Error?.Code ?? "worker-failed";
            }
            if (result.Status is "found" or "failed") stop.Cancel();
        }

        async Task<string?> RunLane(RecoveryWorkerRequest request, int lane)
        {
            void Fault() { lock (gate) failure ??= "worker-protocol"; stop.Cancel(); }
            var outcome = await RunOneAsync(workerPath, dllPath, job, request,
                update => Update(lane, update.CompletedCandidates, update.Stage!, update.PossiblePassword),
                terminal => Terminal(lane, terminal), Fault, stop.Token).ConfigureAwait(false);
            if (outcome is not null) { lock (gate) failure ??= outcome; stop.Cancel(); }
            return outcome;
        }
        var tasks = requests.Select((request, lane) => RunLane(request, lane)).ToArray();
        var all = Task.WhenAll(tasks);
        var cancelSignal = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        using var registration = stop.Token.Register(() => cancelSignal.TrySetResult());
        if (await Task.WhenAny(all, cancelSignal.Task).ConfigureAwait(false) != all)
        {
            // All lanes receive cancellation together; this deadline belongs to the whole pool.
            try { await all.WaitAsync(TimeSpan.FromSeconds(5)).ConfigureAwait(false); }
            catch (TimeoutException)
            {
                job.Terminate();
                await all.ConfigureAwait(false);
            }
        }
        else await all.ConfigureAwait(false);
        var outcomes = await all.ConfigureAwait(false);
        lock (gate)
        {
            var count = completed.Sum();
            if (cancellationToken.IsCancellationRequested) return new("cancelled", count, null, possible);
            failure ??= outcomes.FirstOrDefault(outcome => outcome is not null);
            if (failure is not null) return new("failed", count, null, possible, failure);
            if (found is not null) return new("found", count, found.RecoveredPassword, possible);
            return failure is null ? new("exhausted", count, null, possible) : new("failed", count, null, possible, failure);
        }
    }

    private static async Task<string?> RunOneAsync(string workerPath, string dllPath, ArchiveWorkerJob job,
        RecoveryWorkerRequest request, Action<RecoveryWorkerUpdate> onProgress,
        Action<RecoveryWorkerTerminal> onTerminal, Action onFault, CancellationToken token)
    {
        using var process = new Process { StartInfo = new ProcessStartInfo(Path.GetFullPath(workerPath))
        {
            UseShellExecute = false, CreateNoWindow = true,
            RedirectStandardInput = true, RedirectStandardOutput = true, RedirectStandardError = true,
            StandardInputEncoding = new UTF8Encoding(false), StandardOutputEncoding = Encoding.UTF8,
            StandardErrorEncoding = Encoding.UTF8, WorkingDirectory = Path.GetDirectoryName(Path.GetFullPath(workerPath))!
        } };
        process.StartInfo.ArgumentList.Add("--dll=" + Path.GetFullPath(dllPath));
        Task? stdout = null, stderr = null; RecoveryWorkerTerminal? result = null; string? protocolError = null;
        try
        {
            token.ThrowIfCancellationRequested();
            if (!process.Start()) return "worker-start";
            job.Assign(process);
            process.PriorityClass = ProcessPriorityClass.BelowNormal;
            stdout = DrainAsync(process.StandardOutput, line =>
            {
                try
                {
                    if (line is null) throw new InvalidDataException();
                    using var document = JsonDocument.Parse(line);
                    if (!document.RootElement.TryGetProperty("type", out var type)) throw new InvalidDataException();
                    if (type.GetString() == "recovery-progress")
                    {
                        var update = document.RootElement.Deserialize<RecoveryWorkerUpdate>(Json) ?? throw new InvalidDataException();
                        if (result is not null || update.CompletedCandidates < 0 || update.Stage is not ("testing" or "verifying" or "checking"))
                            throw new InvalidDataException();
                        onProgress(update);
                    }
                    else if (type.GetString() == "result")
                    {
                        if (result is not null) throw new InvalidDataException();
                        var terminal = document.RootElement.Deserialize<RecoveryWorkerTerminal>(Json) ?? throw new InvalidDataException();
                        onTerminal(terminal); result = terminal;
                    }
                    else throw new InvalidDataException();
                }
                catch (Exception exception) when (exception is JsonException or InvalidDataException or ArgumentException or InvalidOperationException)
                { protocolError = "worker-protocol"; onFault(); }
            });
            // No stderr/native text is retained: it may include a candidate or a recovered password.
            stderr = DrainAsync(process.StandardError, _ => { });
            var exit = process.WaitForExitAsync(CancellationToken.None);
            var cancelled = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
            using var registration = token.Register(() => cancelled.TrySetResult());
            await process.StandardInput.WriteLineAsync(JsonSerializer.Serialize(request, Json))
                .WaitAsync(TimeSpan.FromSeconds(30), token).ConfigureAwait(false);
            await process.StandardInput.FlushAsync().ConfigureAwait(false);
            if (await Task.WhenAny(exit, cancelled.Task).ConfigureAwait(false) != exit)
            {
                try { await process.StandardInput.WriteLineAsync("{\"action\":\"cancel\"}").WaitAsync(TimeSpan.FromSeconds(1)).ConfigureAwait(false); }
                catch (Exception e) when (e is IOException or InvalidOperationException or TimeoutException) { }
                // RunStage owns the one five-second deadline and terminates the shared job.
            }
            await exit.ConfigureAwait(false);
            await Task.WhenAll(stdout, stderr).WaitAsync(TimeSpan.FromSeconds(5)).ConfigureAwait(false);
            if (protocolError is not null) return protocolError;
            if (result is null) return token.IsCancellationRequested ? null : "worker-no-result";
            if (result.Status == "cancelled" && !token.IsCancellationRequested) return "worker-unexpected-cancel";
            return null;
        }
        catch (OperationCanceledException) when (token.IsCancellationRequested) { return null; }
        catch (Exception exception) when (exception is IOException or Win32Exception or TimeoutException or InvalidOperationException)
        { return token.IsCancellationRequested ? null : "worker-process"; }
        finally
        {
            try { if (process.Id > 0 && !process.HasExited) { process.Kill(entireProcessTree: true); await process.WaitForExitAsync().ConfigureAwait(false); } }
            catch (InvalidOperationException) { } catch (Win32Exception) { }
            if (stdout is not null && stderr is not null)
                try { await Task.WhenAll(stdout, stderr).WaitAsync(TimeSpan.FromSeconds(5)).ConfigureAwait(false); }
                catch (Exception e) when (e is IOException or TimeoutException or ObjectDisposedException) { }
        }
    }

    private static async Task DrainAsync(StreamReader reader, Action<string?> action)
    {
        var buffer = new char[4096]; var line = new StringBuilder(); var tooLarge = false;
        while (true)
        {
            var count = await reader.ReadAsync(buffer.AsMemory()).ConfigureAwait(false);
            if (count == 0) break;
            for (var i = 0; i < count; i++)
            {
                if (buffer[i] == '\n') { action(tooLarge ? null : line.ToString().TrimEnd('\r')); line.Clear(); tooLarge = false; }
                else if (!tooLarge)
                {
                    if (line.Length == MaximumFrameCharacters) { line.Clear(); tooLarge = true; }
                    else line.Append(buffer[i]);
                }
            }
        }
        if (line.Length != 0 || tooLarge) action(tooLarge ? null : line.ToString().TrimEnd('\r'));
    }

    {
        if (reader.TokenType == JsonTokenType.String && Enum.TryParse<T>(reader.GetString(), true, out var named) && Enum.IsDefined(named)) return named;
        if (reader.TokenType == JsonTokenType.Number && reader.TryGetInt32(out var number))
        {
            var numeric = (T)Enum.ToObject(typeof(T), number);
            if (Enum.IsDefined(numeric)) return numeric;
        }
        reader.Skip();
        return fallback;
    }
    public override void Write(Utf8JsonWriter writer, T value, JsonSerializerOptions options) => writer.WriteStringValue(value.ToString());
}

public sealed class LanguageSetting : JsonConverter<string>
{
    public override bool HandleNull => true;
    public static string Normalize(string? language) => string.Equals(language?.Trim(), "en-US", StringComparison.OrdinalIgnoreCase) ? "en-US" : "zh-CN";
    public static string InstalledDefault()
    {
        try
        {
            using var document = JsonDocument.Parse(File.ReadAllText(Path.Combine(AppContext.BaseDirectory, "installed-language.json")));
            return Normalize(document.RootElement.GetProperty("language").GetString());
        }
        catch (Exception error) when (error is IOException or UnauthorizedAccessException or JsonException or InvalidOperationException or KeyNotFoundException) { return "zh-CN"; }
    }
    public override string Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        if (reader.TokenType == JsonTokenType.String) return Normalize(reader.GetString());
        reader.Skip(); return "zh-CN";
    }
    public override void Write(Utf8JsonWriter writer, string value, JsonSerializerOptions options) => writer.WriteStringValue(Normalize(value));
}

public sealed class ThemePreferenceSettingConverter : JsonConverter<string>
{
    public override bool HandleNull => true;
    public static string Normalize(string? value) => value?.Trim().ToLowerInvariant() switch
    {
        "light" => "Light", "dark" => "Dark", _ => "System"
    };
    public override string Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        if (reader.TokenType == JsonTokenType.String) return Normalize(reader.GetString());
        reader.Skip();
        return "System";
    }
    public override void Write(Utf8JsonWriter writer, string value, JsonSerializerOptions options)
        => writer.WriteStringValue(Normalize(value));
}

public sealed class GifQualitySettingConverter : JsonConverter<GifQualityLevel>
{
    public override GifQualityLevel Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        if (reader.TokenType == JsonTokenType.String && Enum.TryParse<GifQualityLevel>(reader.GetString(), true, out var named) && Enum.IsDefined(named)) return named;
        if (reader.TokenType == JsonTokenType.Number && reader.TryGetInt32(out var number) && Enum.IsDefined((GifQualityLevel)number)) return (GifQualityLevel)number;
        reader.Skip();
        return GifQualityLevel.Medium;
    }
    public override void Write(Utf8JsonWriter writer, GifQualityLevel value, JsonSerializerOptions options) => writer.WriteStringValue(value.ToString());
}

public sealed class SettingsStore
{
    private readonly object gate = new();
    private readonly string directory;
    private readonly JsonSerializerOptions options = new() { WriteIndented = true, Converters = { new JsonStringEnumConverter() } };
    public string FilePath => Path.Combine(directory, "settings.json");
    public SettingsStore(string directory) => this.directory = Path.GetFullPath(directory);

    public AppSettings Load()
    {
        lock (gate)
        {
            try { return LoadFrom(FilePath); }
            catch (IOException) { }
            catch (UnauthorizedAccessException) { }
            catch (JsonException) { }
            try { return LoadFrom(FilePath + ".bak"); }
            catch (IOException) { return new AppSettings(); }
            catch (UnauthorizedAccessException) { return new AppSettings(); }
            catch (JsonException) { return new AppSettings(); }
        }
    }

    private AppSettings LoadFrom(string path)
    {
        using var document = JsonDocument.Parse(File.ReadAllText(path));
        if (document.RootElement.ValueKind != JsonValueKind.Object) throw new JsonException("Settings must be a JSON object.");
        var settings = JsonSerializer.Deserialize<AppSettings>(document.RootElement, options) ?? new AppSettings();
        // A pre-language user profile already used Chinese; an installer language must not overwrite it.
        settings.Language = document.RootElement.TryGetProperty("Language", out _) ? LanguageSetting.Normalize(settings.Language) : "zh-CN";
        if (string.IsNullOrWhiteSpace(settings.OutputDirectory)) settings.OutputDirectory = new AppSettings().OutputDirectory;
        settings.PresetIds ??= new();
        settings.ThemePreference = ThemePreferenceSettingConverter.Normalize(settings.ThemePreference);
        if (!Enum.IsDefined(settings.MotionPreference)) settings.MotionPreference = MotionPreference.System;
        if (settings.CompressionMode is not (CompressionMode.Standard or CompressionMode.TargetSize)) settings.CompressionMode = CompressionMode.Standard;
        if (!Enum.IsDefined(settings.CompressionLevel)) settings.CompressionLevel = CompressionLevel.Medium;
        if (!Enum.IsDefined(settings.GifQuality)) settings.GifQuality = GifQualityLevel.Medium;
        if (settings.CompressionMaximumMb < .000001m || settings.CompressionMaximumMb > long.MaxValue / 1000000m) settings.CompressionMaximumMb = 10;
        if (settings.ArchiveSplitMiB < 1m || settings.ArchiveSplitMiB > 102400m) settings.ArchiveSplitMiB = 100;
        if (settings.ArchiveMaximumEntries < 1) settings.ArchiveMaximumEntries = 100_000;
        if (settings.ArchiveMaximumDepth < 1) settings.ArchiveMaximumDepth = 64;
        if (settings.ArchiveMaximumBytes < 1) settings.ArchiveMaximumBytes = 100L * 1024 * 1024 * 1024;
        if (settings.RecoveryDigits is not (4 or 6)) settings.RecoveryDigits = 4;
        if (!Enum.IsDefined(settings.ImageObfuscationMode)) settings.ImageObfuscationMode = ImageObfuscationMode.Gilbert;
        if (settings.ImageObfuscationIterations is < 1 or > 10) settings.ImageObfuscationIterations = 1;
        settings.Pet ??= new();
        if (document.RootElement.ValueKind != JsonValueKind.Object || !document.RootElement.TryGetProperty("Pet", out var petDocument) ||
            petDocument.ValueKind != JsonValueKind.Object || !petDocument.TryGetProperty("MotionMode", out _))
            settings.Pet.MotionMode = settings.MotionPreference == MotionPreference.Reduced ? PetMotionMode.Reduced : PetMotionMode.Gentle;
        settings.Pet.Validate();
        return settings;
    }

    public void Save(AppSettings settings)
    {
        ArgumentNullException.ThrowIfNull(settings);
        lock (gate)
        {
            settings.ThemePreference = ThemePreferenceSettingConverter.Normalize(settings.ThemePreference);
            settings.Language = LanguageSetting.Normalize(settings.Language);
            settings.Pet ??= new();
            settings.Pet.Validate();
            Directory.CreateDirectory(directory);
            var temporary = Path.Combine(directory, $"settings-{Guid.NewGuid():N}.tmp");
            try
            {
                using (var stream = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None, 4096, FileOptions.WriteThrough))
                {
                    var bytes = Encoding.UTF8.GetBytes(JsonSerializer.Serialize(settings, options));
                    stream.Write(bytes);
                    stream.Flush(flushToDisk: true);
                }
                // Rename with replacement is supported by the portable app's restricted
                // filesystem; ReplaceFile can require additional metadata permissions.
                if (File.Exists(FilePath)) File.Copy(FilePath, FilePath + ".bak", overwrite: true);
                File.Move(temporary, FilePath, overwrite: true);
            }
            finally
            {
                // An already committed save stays successful even when cleanup is denied.
                try { if (File.Exists(temporary)) File.Delete(temporary); }
                catch (IOException) { }
                catch (UnauthorizedAccessException) { }
            }
        }
    }
}
